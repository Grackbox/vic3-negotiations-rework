# NR: the Industrialists' heavy industry promise (options 5 and 6, promise_quest_type 103): X levels of a heavy
# industry type staffed at least 90%, profitable and not subsidized, held 6 months in a row, within 10 years (+3 with the
# vanilla extension).

je_nr_industrialists_healthy = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_internal_affairs

	scripted_button = negotiation_go_away_button
	scripted_button = negotiation_extension_button

	immediate = {
		random_interest_group = {
			limit = {
				has_variable = ig_new_quest_starting
				has_variable = promise_quest_type
				var:promise_quest_type = 103 # NR: heavy industry promise
			}
			save_scope_as = ig
			remove_variable = ig_new_quest_starting
		}
		set_variable = { # copies for the progress bar, which cannot see scope:ig
			name = nr_industrialists_healthy_type_c
			value = scope:ig.var:nr_industrialists_healthy_type
		}
		set_variable = {
			name = nr_industrialists_healthy_target_c
			value = scope:ig.var:nr_industrialists_healthy_target
		}
	}

	scripted_progress_bar = nr_industrialists_healthy_bar

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_nr_industrialists_healthy_status
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = nr_industrialists_healthy_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(nr_industrialists_healthy_bar)" >= 6
			}
		}
	}

	on_complete = {
		if = {
			limit = { scope:ig ?= { has_variable = promise_quest_degree } }
			promise_quest_completed = yes
		}
		nr_industrialists_healthy_cleanup = yes
	}

	fail = {
		OR = {
			is_promise_abandoned = yes
			custom_tooltip = {
				text = time_runs_out_tt
				scope:journal_entry.journal_entry_age >= {
					value = 3650 # 10 years
					if = {
						limit = {
							scope:journal_entry = {
								has_variable = asked_for_extension_var
							}
						}
						add = 1095
					}
				}
			}
		}
	}

	on_fail = {
		if = {
			limit = { scope:ig ?= { has_variable = promise_quest_type } }
			promise_quest_failed = yes
		}
		nr_industrialists_healthy_cleanup = yes
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
