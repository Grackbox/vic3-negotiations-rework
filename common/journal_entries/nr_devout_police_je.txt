# NR: the clergy's petition for a police law (feature devout_police). The law is chosen when the petition is offered
# (nr_devout_police_pick_target) and kept in nr_devout_police_petition_law (1 / 2 / 3).
# 4 years: while that law is being enacted it gets +25% enactment speed.
# Complete: that law is active. Timeout: Devout -2 approval for 5 years.

je_nr_devout_police_petition = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	immediate = {
		ig:ig_devout ?= { save_scope_as = ig }
		nr_devout_police_petition_boost = yes
	}

	on_monthly_pulse = {
		effect = {
			nr_devout_police_petition_boost = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_enacting_1
				trigger = {
					var:nr_devout_police_petition_law ?= 1
					is_enacting_law = law_type:law_local_police
				}
			}
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_enacting_2
				trigger = {
					var:nr_devout_police_petition_law ?= 2
					is_enacting_law = law_type:law_dedicated_police
				}
			}
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_enacting_3
				trigger = {
					var:nr_devout_police_petition_law ?= 3
					is_enacting_law = law_type:law_militarized_police
				}
			}
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_waiting_1
				trigger = { var:nr_devout_police_petition_law ?= 1 }
			}
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_waiting_2
				trigger = { var:nr_devout_police_petition_law ?= 2 }
			}
			triggered_desc = {
				desc = je_nr_devout_police_petition_status_waiting_3
				trigger = { var:nr_devout_police_petition_law ?= 3 }
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_nr_devout_police_petition_complete_tt
			OR = {
				AND = {
					var:nr_devout_police_petition_law ?= 1
					has_law = law_type:law_local_police
				}
				AND = {
					var:nr_devout_police_petition_law ?= 2
					has_law = law_type:law_dedicated_police
				}
				AND = {
					var:nr_devout_police_petition_law ?= 3
					has_law = law_type:law_militarized_police
				}
			}
		}
	}

	on_complete = {
		nr_petition_end = { PETITION = nr_devout_police_petition }
		hidden_effect = { nr_devout_police_petition_cleanup = yes }
	}

	on_timeout = {
		nr_petition_timeout = { PETITION = nr_devout_police_petition }
		hidden_effect = { nr_devout_police_petition_cleanup = yes }
	}

	timeout = 1461 # 4 years

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
