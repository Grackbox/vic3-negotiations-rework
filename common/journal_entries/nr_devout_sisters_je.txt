# NR: the orders' petition for Charity Hospitals (feature devout_sisters).
# 4 years: while Charity Hospitals is being enacted it gets +25% enactment speed.
# Complete: Charity Hospitals is the law. Timeout: Devout -2 approval for 5 years.

je_nr_devout_sisters_petition = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	immediate = {
		ig:ig_devout ?= { save_scope_as = ig }
		nr_petition_boost = { PETITION = nr_devout_sisters_petition LAW = law_charitable_health_system }
	}

	on_monthly_pulse = {
		effect = {
			nr_petition_boost = { PETITION = nr_devout_sisters_petition LAW = law_charitable_health_system }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_nr_devout_sisters_petition_status_enacting
				trigger = { is_enacting_law = law_type:law_charitable_health_system }
			}
			triggered_desc = {
				desc = je_nr_devout_sisters_petition_status_waiting
			}
		}
	}

	complete = {
		has_law = law_type:law_charitable_health_system
	}

	on_complete = {
		nr_petition_end = { PETITION = nr_devout_sisters_petition }
	}

	on_timeout = {
		nr_petition_timeout = { PETITION = nr_devout_sisters_petition }
	}

	timeout = 1461 # 4 years

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
