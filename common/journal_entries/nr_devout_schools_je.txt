# NR: the clergy's petition for Religious Schools (feature devout_schools).
# 4 years: while Religious Schools is being enacted it gets +25% enactment speed.
# Complete: Religious Schools is the law. Timeout: Devout -2 approval for 5 years.

je_nr_devout_schools_petition = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	immediate = {
		ig:ig_devout ?= { save_scope_as = ig }
		nr_devout_schools_petition_boost = yes
	}

	on_monthly_pulse = {
		effect = {
			nr_devout_schools_petition_boost = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_nr_devout_schools_petition_status_enacting
				trigger = { is_enacting_law = law_type:law_religious_schools }
			}
			triggered_desc = {
				desc = je_nr_devout_schools_petition_status_waiting
			}
		}
	}

	complete = {
		has_law = law_type:law_religious_schools
	}

	on_complete = {
		hidden_effect = { remove_variable = nr_devout_schools_petition_boosted }
	}

	on_timeout = {
		scope:ig ?= {
			add_modifier = {
				name = nr_devout_schools_petition_ignored
				days = normal_modifier_time
			}
		}
		hidden_effect = { remove_variable = nr_devout_schools_petition_boosted }
	}

	timeout = 1461 # 4 years

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
