# NR: Devout church officials (negotiation option 2, replaces vanilla "places in the administration" for the Devout).
# The form (where the officials go) is chosen by event nr_devout_officials.1 a week later.

# Negotiation option. Scope: country, scope:ig = Devout.
nr_devout_officials_start = {
	custom_tooltip = nr_devout_officials_tt
	hidden_effect = {
		set_variable = {
			name = nr_devout_officials_degree
			value = scope:ig.amenability_level
		}
		nr_campaign_take_slot = { KEY = nr_devout_officials }
	}
	# penalty like vanilla negotiation_bureaucracy (-7.5% bureaucracy x level 1 / 2 / 4), rescaled by the form a week later
	nr_devout_officials_set_penalty = { FACTOR = 1 }
	scope:ig = {
		finish_negotiation = "negotiation.1.o2"
		add_modifier = {
			name = nr_devout_officials_benefits
			days = long_modifier_time
			multiplier = nr_neg_level_scale # 1 / 2 / 3: +12 / 24 / 36% political strength
			is_decaying = yes
		}
		improve_law_stance = yes
	}
	trigger_event = {
		id = nr_devout_officials.1
		days = 7 # delay of the form event
		popup = yes
	}
}

# Offers one more random form in event nr_devout_officials.1 (sets nr_devout_officials_offer_<form>). Scope: country.
nr_devout_officials_offer_random_form = {
	random_list = {
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = censorship } }
			set_variable = nr_devout_officials_offer_censorship
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = registers } }
			set_variable = nr_devout_officials_offer_registers
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = slaves } }
			set_variable = nr_devout_officials_offer_slaves
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = church_tax } }
			set_variable = nr_devout_officials_offer_church_tax
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = prisons } }
			set_variable = nr_devout_officials_offer_prisons
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = calendar } }
			set_variable = nr_devout_officials_offer_calendar
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = seminarists } }
			set_variable = nr_devout_officials_offer_seminarists
		}
		1 = {
			trigger = { nr_devout_officials_can_offer = { FORM = synod } }
			set_variable = nr_devout_officials_offer_synod
		}
	}
}

# Chosen form: its modifier for 10 years. Scope: country. $FORM$ = censorship / registers / ...
# Forms with a penalty other than x1 also call nr_devout_officials_set_penalty.
nr_devout_officials_choose_form = {
	add_modifier = {
		name = nr_devout_officials_$FORM$
		days = long_modifier_time
		is_decaying = yes
	}
}

# Sets the bureaucracy penalty of this campaign (its own slot modifier nr_devout_officials_penalty_<slot>)
# to level x $FACTOR$ (1 at the negotiation; 0.75 / 1.25 / 1.5 for the form). Scope: country.
nr_devout_officials_set_penalty = {
	if = {
		limit = { has_variable = nr_devout_officials_current_slot }
		if = {
			limit = { var:nr_devout_officials_current_slot = 1 }
			nr_devout_officials_set_slot_penalty = { SLOT = 1 FACTOR = $FACTOR$ }
		}
		else_if = {
			limit = { var:nr_devout_officials_current_slot = 2 }
			nr_devout_officials_set_slot_penalty = { SLOT = 2 FACTOR = $FACTOR$ }
		}
		else = {
			nr_devout_officials_set_slot_penalty = { SLOT = 3 FACTOR = $FACTOR$ }
		}
	}
}

nr_devout_officials_set_slot_penalty = {
	if = {
		limit = { has_modifier = nr_devout_officials_penalty_$SLOT$ }
		remove_modifier = nr_devout_officials_penalty_$SLOT$
	}
	add_modifier = {
		name = nr_devout_officials_penalty_$SLOT$
		multiplier = {
			value = nr_devout_officials_level
			multiply = $FACTOR$
		}
		days = long_modifier_time
		is_decaying = yes
	}
}

# Clears the data of event nr_devout_officials.1. Scope: country.
nr_devout_officials_cleanup = {
	remove_variable = nr_devout_officials_degree
	remove_variable = nr_devout_officials_current_slot
	remove_variable = nr_devout_officials_offer_censorship
	remove_variable = nr_devout_officials_offer_registers
	remove_variable = nr_devout_officials_offer_slaves
	remove_variable = nr_devout_officials_offer_church_tax
	remove_variable = nr_devout_officials_offer_prisons
	remove_variable = nr_devout_officials_offer_calendar
	remove_variable = nr_devout_officials_offer_seminarists
	remove_variable = nr_devout_officials_offer_synod
}

# Censorship Committees end when Protected Speech is enacted (called from nr_on_law_activated). Scope: country.
nr_devout_officials_check_censorship = {
	if = {
		limit = {
			has_modifier = nr_devout_officials_censorship
			has_law = law_type:law_protected_speech
		}
		remove_modifier = nr_devout_officials_censorship
		if = {
			limit = { has_variable = nr_devout_censorship_campaign }
			remove_variable = nr_devout_censorship_campaign # no follow-up event
		}
	}
}
