# NR: Sisters of Mercy (feature devout_sisters).

# Gives the Charity Hospitals enactment +25% speed once per enactment. Scope: country.
nr_devout_sisters_petition_boost = {
	if = {
		limit = {
			is_enacting_law = law_type:law_charitable_health_system
			NOT = { has_variable = nr_devout_sisters_petition_boosted }
		}
		add_enactment_modifier = {
			name = nr_devout_sisters_petition_speed
		}
		set_variable = nr_devout_sisters_petition_boosted
	}
	else_if = {
		limit = {
			NOT = { is_enacting_law = law_type:law_charitable_health_system }
			has_variable = nr_devout_sisters_petition_boosted
		}
		remove_variable = nr_devout_sisters_petition_boosted # enactment ended or failed: boost the next attempt again
	}
}
