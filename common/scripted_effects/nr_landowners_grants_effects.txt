# NR: landowners' grants (negotiation option 1, feature landowners_grants).
# The form is chosen by event nr_landowners_grants.1 a week later.

# Negotiation option. Scope: country, scope:ig = Landowners.
nr_landowners_grants_start = {
	custom_tooltip = nr_landowners_grants_tt
	hidden_effect = {
		# Blocks a second option 1 of the Landowners until the form event fires (nr_softbribe_option_allowed)
		set_variable = {
			name = nr_landowners_grants_pending
			days = 14
		}
		set_variable = {
			name = nr_landowners_grants_cost_amount
			value = neg_bribe_amount
		}
		set_variable = {
			name = nr_landowners_grants_degree
			value = scope:ig.amenability_level
		}
		nr_campaign_take_slot = { KEY = nr_landowners_grants }
	}
	custom_tooltip = nr_softbribe_cost_tt # the expenses modifier is set from hidden variables, so the tooltip would not show it
	nr_softbribe_set_cost = { KEY = nr_landowners_grants FACTOR = 1 }
	scope:ig = {
		finish_negotiation = "negotiation.1.o1"
		add_modifier = {
			name = nr_landowners_grants_favor
			days = long_modifier_time
			multiplier = nr_neg_level_scale
			is_decaying = yes
		}
		improve_law_stance = yes
	}
	trigger_event = {
		id = nr_landowners_grants.1
		days = 7 # delay of the form event
		popup = yes
	}
}

# Offers one more random form in event nr_landowners_grants.1 (sets nr_landowners_grants_offer_<form>). Scope: country.
nr_landowners_grants_offer_random_form = {
	random_list = {
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = crown_lands } }
			set_variable = nr_landowners_grants_offer_crown_lands
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = tax_exemptions } }
			set_variable = nr_landowners_grants_offer_tax_exemptions
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = corn_laws } }
			set_variable = nr_landowners_grants_offer_corn_laws
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = land_bank } }
			set_variable = nr_landowners_grants_offer_land_bank
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = corvee } }
			set_variable = nr_landowners_grants_offer_corvee
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = redemption } }
			set_variable = nr_landowners_grants_offer_redemption
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = fugitives } }
			set_variable = nr_landowners_grants_offer_fugitives
		}
		1 = {
			trigger = { nr_landowners_grants_can_offer = { FORM = slave_import } }
			set_variable = nr_landowners_grants_offer_slave_import
		}
	}
}

# Chosen form: its modifier for 10 years, its expenses (x $FACTOR$), the slot it occupies (for an interruption),
# the running mark for its follow-up. Scope: country.
nr_landowners_grants_choose_form = {
	add_modifier = {
		name = nr_landowners_grants_$FORM$
		days = long_modifier_time
		is_decaying = yes
	}
	nr_softbribe_set_cost = { KEY = nr_landowners_grants FACTOR = $FACTOR$ }
	hidden_effect = {
		if = {
			limit = { has_variable = nr_landowners_grants_current_slot }
			set_variable = {
				name = nr_landowners_grants_$FORM$_slot
				value = var:nr_landowners_grants_current_slot
				days = long_modifier_time
			}
		}
		set_variable = {
			name = nr_landowners_grants_$FORM$_running
			days = 3660
		}
	}
	nr_landowners_grants_movements = yes
}

# Movement radicalism of the running forms (and of the Free Import amendment); on choice and yearly. Scope: country.
nr_landowners_grants_movements = {
	if = {
		limit = { has_modifier = nr_landowners_grants_corvee }
		nr_movement_radicalism = { MOVEMENT = movement_land_reform MOD = nr_movement_radicalism_up_010 }
	}
	if = {
		limit = { has_modifier = nr_landowners_grants_redemption }
		nr_movement_radicalism = { MOVEMENT = movement_land_reform MOD = nr_movement_radicalism_down_015 }
	}
	if = {
		limit = { has_modifier = nr_landowners_grants_fugitives }
		nr_movement_radicalism = { MOVEMENT = movement_anti_slavery MOD = nr_movement_radicalism_up_015 }
		nr_movement_radicalism = { MOVEMENT = movement_pro_slavery MOD = nr_movement_radicalism_down_015 }
	}
	if = {
		limit = { has_modifier = nr_landowners_grants_slave_import }
		nr_movement_radicalism = { MOVEMENT = movement_anti_slavery MOD = nr_movement_radicalism_up_010 }
		nr_movement_radicalism = { MOVEMENT = movement_pro_slavery MOD = nr_movement_radicalism_down_010 }
	}
	if = {
		limit = { any_active_law = { has_amendment = amendment_type:amendment_nr_landowners_free_import } }
		nr_movement_radicalism = { MOVEMENT = movement_pro_slavery MOD = nr_movement_radicalism_down_010 }
	}
}

# Clears the data of event nr_landowners_grants.1. Scope: country.
nr_landowners_grants_cleanup = {
	remove_variable = nr_landowners_grants_pending
	remove_variable = nr_landowners_grants_cost_amount
	remove_variable = nr_landowners_grants_degree
	remove_variable = nr_landowners_grants_current_slot
	remove_variable = nr_landowners_grants_offer_crown_lands
	remove_variable = nr_landowners_grants_offer_tax_exemptions
	remove_variable = nr_landowners_grants_offer_corn_laws
	remove_variable = nr_landowners_grants_offer_land_bank
	remove_variable = nr_landowners_grants_offer_corvee
	remove_variable = nr_landowners_grants_offer_redemption
	remove_variable = nr_landowners_grants_offer_fugitives
	remove_variable = nr_landowners_grants_offer_slave_import
}

# A law change ends a form (called from nr_on_law_activated). Scope: country.
nr_landowners_grants_check_laws = {
	if = {
		limit = {
			has_modifier = nr_landowners_grants_corn_laws
			has_law = law_type:law_free_trade
		}
		nr_landowners_grants_interrupt = { FORM = corn_laws EVENT = 21 }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_grants_corvee
			NOR = {
				has_law = law_type:law_serfdom
				has_law = law_type:law_manorialism
			}
		}
		nr_landowners_grants_interrupt = { FORM = corvee EVENT = 22 }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_grants_fugitives
			has_law = law_type:law_slavery_banned
		}
		nr_landowners_grants_interrupt = { FORM = fugitives EVENT = 23 }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_grants_slave_import
			NOT = { has_law = law_type:law_slave_trade }
		}
		nr_landowners_grants_interrupt = { FORM = slave_import EVENT = 24 }
	}
}

# Ends a form early: its modifier, its expenses and its slot go, no follow-up; popup event $EVENT$ tells what happened
# and applies the Landowners' reaction. Scope: country.
nr_landowners_grants_interrupt = {
	remove_modifier = nr_landowners_grants_$FORM$
	hidden_effect = {
		if = {
			limit = { var:nr_landowners_grants_$FORM$_slot ?= 1 }
			nr_landowners_grants_free_slot = { SLOT = 1 }
		}
		else_if = {
			limit = { var:nr_landowners_grants_$FORM$_slot ?= 2 }
			nr_landowners_grants_free_slot = { SLOT = 2 }
		}
		else_if = {
			limit = { var:nr_landowners_grants_$FORM$_slot ?= 3 }
			nr_landowners_grants_free_slot = { SLOT = 3 }
		}
		if = {
			limit = { has_variable = nr_landowners_grants_$FORM$_slot }
			remove_variable = nr_landowners_grants_$FORM$_slot
		}
		if = {
			limit = { has_variable = nr_landowners_grants_$FORM$_running }
			remove_variable = nr_landowners_grants_$FORM$_running
		}
	}
	trigger_event = { id = nr_landowners_grants.$EVENT$ popup = yes } # tells what happened; the Landowners' reaction is in its options
}


nr_landowners_grants_free_slot = {
	if = {
		limit = { has_modifier = nr_landowners_grants_expenses_$SLOT$ }
		remove_modifier = nr_landowners_grants_expenses_$SLOT$
	}
	if = {
		limit = { has_variable = nr_landowners_grants_slot_$SLOT$ }
		remove_variable = nr_landowners_grants_slot_$SLOT$
	}
}
