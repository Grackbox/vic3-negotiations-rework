# NR: campaign slots - how many 10-year campaigns of one kind a group runs at the same time.
# A campaign takes a slot (country variable <KEY>_slot_1..3, expires with the campaign) and remembers it
# in <KEY>_current_slot until its form event is done. Per-slot modifiers keep campaigns from overwriting each other.
# The limit by the group's clout is checked by nr_campaign_slot_free (scripted_triggers/nr_campaign_slots.txt).
# Parameters: KEY = variable prefix (nr_devout_softbribe / nr_devout_officials). Scope: country.

# Takes the first free slot for long_modifier_time and stores its number in <KEY>_current_slot.
nr_campaign_take_slot = {
	if = {
		limit = { NOT = { has_variable = $KEY$_slot_1 } }
		set_variable = {
			name = $KEY$_slot_1
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 1
		}
	}
	else_if = {
		limit = { NOT = { has_variable = $KEY$_slot_2 } }
		set_variable = {
			name = $KEY$_slot_2
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 2
		}
	}
	else = {
		set_variable = {
			name = $KEY$_slot_3
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 3
		}
	}
}

# ---------- Campaign interruption (pattern in README_NR.md) ----------
# A law change ends a campaign early: a popup event tells what happened, and its option applies the group's reaction.
# $IG$ = interest group type. Scope: country.

# The group is in government: it gave the concession up itself.
nr_campaign_react_gave_up = {
	ig:$IG$ ?= {
		add_modifier = {
			name = nr_campaign_gave_up
			days = normal_modifier_time
		}
	}
}

# The group is not in government: the order was broken over its head - marginalization and a backlash
# ($BACKLASH$ = reaction / planters / none).
nr_campaign_react_overruled = {
	ig:$IG$ ?= {
		add_modifier = {
			name = nr_campaign_overruled
			days = normal_modifier_time
		}
		add_modifier = {
			name = nr_campaign_marginalized
			days = normal_modifier_time
			is_decaying = yes
		}
	}
	nr_campaign_backlash_$BACKLASH$ = yes
}

# Backlash: the old order gathers supporters.
nr_campaign_backlash_reaction = {
	add_modifier = {
		name = nr_campaign_reaction
		days = normal_modifier_time
	}
}

# Backlash: the planters are up in arms.
nr_campaign_backlash_planters = {
	nr_movement_radicalism = { MOVEMENT = movement_pro_slavery MOD = nr_movement_radicalism_up_020 }
}

nr_campaign_backlash_none = {
	# no backlash
}

# ---------- Campaign forms (shared by every campaign with a form event) ----------
# KEY = campaign key (nr_landowners_offices, ...); the form modifier is <KEY>_<FORM>. Scope: country.

# The chosen form: its modifier for 10 years, decaying.
nr_campaign_choose_form = {
	add_modifier = {
		name = $KEY$_$FORM$
		days = long_modifier_time
		is_decaying = yes
	}
}

# Running mark of the form, required by its follow-up event 10 years later.
nr_campaign_mark_running = {
	hidden_effect = {
		set_variable = {
			name = $KEY$_$FORM$_running
			days = 3660
		}
	}
}

# The slot the form occupies, for an interruption (only forms that a law change can end).
nr_campaign_remember_slot = {
	hidden_effect = {
		if = {
			limit = { has_variable = $KEY$_current_slot }
			set_variable = {
				name = $KEY$_$FORM$_slot
				value = var:$KEY$_current_slot
				days = long_modifier_time
			}
		}
	}
}

# Bureaucracy penalty of this campaign (modifier <KEY>_penalty_<slot>, -7.5% bureaucracy per point like vanilla
# negotiation_bureaucracy) = negotiation level (1 / 2 / 4) x $FACTOR$.
nr_campaign_set_penalty = {
	if = {
		limit = { has_variable = $KEY$_current_slot }
		if = {
			limit = { var:$KEY$_current_slot = 1 }
			nr_campaign_set_slot_penalty = { KEY = $KEY$ SLOT = 1 FACTOR = $FACTOR$ }
		}
		else_if = {
			limit = { var:$KEY$_current_slot = 2 }
			nr_campaign_set_slot_penalty = { KEY = $KEY$ SLOT = 2 FACTOR = $FACTOR$ }
		}
		else = {
			nr_campaign_set_slot_penalty = { KEY = $KEY$ SLOT = 3 FACTOR = $FACTOR$ }
		}
	}
}

nr_campaign_set_slot_penalty = {
	if = {
		limit = { has_modifier = $KEY$_penalty_$SLOT$ }
		remove_modifier = $KEY$_penalty_$SLOT$
	}
	add_modifier = {
		name = $KEY$_penalty_$SLOT$
		multiplier = {
			value = 1
			if = {
				limit = { exists = scope:ig }
				value = scope:ig.amenability_level
			}
			if = {
				limit = { has_variable = $KEY$_degree }
				value = var:$KEY$_degree
			}
			multiply = $FACTOR$
		}
		days = long_modifier_time
		is_decaying = yes
	}
}

# A law change ends a form early: its modifier, its cost ($COST$ = penalty / expenses) and its slot go, no follow-up;
# popup event <KEY>.$EVENT$ tells what happened and applies the group's reaction (pattern "Campaign interruption").
nr_campaign_interrupt = {
	remove_modifier = $KEY$_$FORM$
	hidden_effect = {
		if = {
			limit = { var:$KEY$_$FORM$_slot ?= 1 }
			nr_campaign_free_slot = { KEY = $KEY$ SLOT = 1 COST = $COST$ }
		}
		else_if = {
			limit = { var:$KEY$_$FORM$_slot ?= 2 }
			nr_campaign_free_slot = { KEY = $KEY$ SLOT = 2 COST = $COST$ }
		}
		else_if = {
			limit = { var:$KEY$_$FORM$_slot ?= 3 }
			nr_campaign_free_slot = { KEY = $KEY$ SLOT = 3 COST = $COST$ }
		}
		if = {
			limit = { has_variable = $KEY$_$FORM$_slot }
			remove_variable = $KEY$_$FORM$_slot
		}
		if = {
			limit = { has_variable = $KEY$_$FORM$_running }
			remove_variable = $KEY$_$FORM$_running
		}
	}
	trigger_event = { id = $KEY$.$EVENT$ popup = yes }
}

nr_campaign_free_slot = {
	if = {
		limit = { has_modifier = $KEY$_$COST$_$SLOT$ }
		remove_modifier = $KEY$_$COST$_$SLOT$
	}
	if = {
		limit = { has_variable = $KEY$_slot_$SLOT$ }
		remove_variable = $KEY$_slot_$SLOT$
	}
}
