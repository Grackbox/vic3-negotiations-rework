# NR: campaign slots - how many 10-year campaigns of one kind a group runs at the same time.
# A campaign takes a slot (country variable <KEY>_slot_1..3, expires with the campaign) and remembers it
# in <KEY>_current_slot until its form event is done. Per-slot modifiers keep campaigns from overwriting each other.
# The limit by the group's clout is checked by nr_campaign_slot_free (scripted_triggers/nr_campaign_slots.txt).
# Parameters: KEY = variable prefix (nr_devout_softbribe / nr_devout_officials). Scope: country.

# Takes the first free slot for long_modifier_time and stores its number in <KEY>_current_slot.
nr_campaign_take_slot = {
	if = {
		limit = { NOT = { has_variable = $KEY$_slot_1 } }
		set_variable = {
			name = $KEY$_slot_1
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 1
		}
	}
	else_if = {
		limit = { NOT = { has_variable = $KEY$_slot_2 } }
		set_variable = {
			name = $KEY$_slot_2
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 2
		}
	}
	else = {
		set_variable = {
			name = $KEY$_slot_3
			days = long_modifier_time
		}
		set_variable = {
			name = $KEY$_current_slot
			value = 3
		}
	}
}

# ---------- Campaign interruption (pattern in README_NR.md) ----------
# A law change ends a campaign early: a popup event tells what happened, and its option applies the group's reaction.
# $IG$ = interest group type. Scope: country.

# The group is in government: it gave the concession up itself.
nr_campaign_react_gave_up = {
	ig:$IG$ ?= {
		add_modifier = {
			name = nr_campaign_gave_up
			days = normal_modifier_time
		}
	}
}

# The group is not in government: the order was broken over its head - marginalization and a backlash
# ($BACKLASH$ = reaction / planters / none).
nr_campaign_react_overruled = {
	ig:$IG$ ?= {
		add_modifier = {
			name = nr_campaign_overruled
			days = normal_modifier_time
		}
		add_modifier = {
			name = nr_campaign_marginalized
			days = normal_modifier_time
			is_decaying = yes
		}
	}
	nr_campaign_backlash_$BACKLASH$ = yes
}

# Backlash: the old order gathers supporters.
nr_campaign_backlash_reaction = {
	add_modifier = {
		name = nr_campaign_reaction
		days = normal_modifier_time
	}
}

# Backlash: the planters are up in arms.
nr_campaign_backlash_planters = {
	nr_movement_radicalism = { MOVEMENT = movement_pro_slavery MOD = nr_movement_radicalism_up_020 }
}

nr_campaign_backlash_none = {
	# no backlash
}
