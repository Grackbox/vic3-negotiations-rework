# NR: the Landowners' places in the provinces (negotiation option 2 for the Landowners, feature landowners_offices).
# The form (which posts they get) is chosen by event nr_landowners_offices.1 a week later. Same scheme as nr_devout_officials.

# Negotiation option. Scope: country, scope:ig = Landowners.
nr_landowners_offices_start = {
	custom_tooltip = nr_landowners_offices_tt
	hidden_effect = {
		set_variable = {
			name = nr_landowners_offices_degree
			value = scope:ig.amenability_level
		}
		nr_campaign_take_slot = { KEY = nr_landowners_offices }
	}
	custom_tooltip = nr_campaign_penalty_tt # the penalty is set from hidden variables, so the tooltip would not show it
	nr_campaign_set_penalty = { KEY = nr_landowners_offices FACTOR = 1 } # like vanilla negotiation_bureaucracy, rescaled by the form a week later
	scope:ig = {
		finish_negotiation = "negotiation.1.o2"
		add_modifier = {
			name = nr_landowners_offices_benefits
			days = long_modifier_time
			multiplier = nr_neg_level_scale
			is_decaying = yes
		}
		improve_law_stance = yes
	}
	trigger_event = {
		id = nr_landowners_offices.1
		days = 7
		popup = yes
	}
}

# Clears the data of event nr_landowners_offices.1. Scope: country.

	remove_variable = nr_landowners_offices_offer_patrimonial
	remove_variable = nr_landowners_offices_offer_justices

	remove_variable = nr_landowners_offices_offer_chancery

# A law change ends a form (called from nr_on_law_activated). Scope: country.
nr_landowners_offices_check_laws = {
	if = {
		limit = {
			has_modifier = nr_landowners_offices_patrimonial
			NOR = { has_law = law_type:law_serfdom has_law = law_type:law_manorialism has_law = law_type:law_tenant_farmers has_law = law_type:law_latifundias has_law = law_type:law_expanded_latifundias }
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = patrimonial EVENT = 31 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_constabulary
			OR = { has_law = law_type:law_dedicated_police has_law = law_type:law_militarized_police }
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = constabulary EVENT = 32 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_assemblies
			has_law = law_type:law_universal_suffrage
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = assemblies EVENT = 33 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_tax_farming
			OR = { has_law = law_type:law_proportional_taxation has_law = law_type:law_graduated_taxation }
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = tax_farming EVENT = 34 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_patrols
			has_law = law_type:law_slavery_banned
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = patrols EVENT = 35 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_state_slaves
			has_law = law_type:law_slavery_banned
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = state_slaves EVENT = 36 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_officers
			OR = { has_law = law_type:law_national_militia has_law = law_type:law_mass_conscription }
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = officers EVENT = 37 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_recruits
			NOT = { has_law = law_type:law_peasant_levies }
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = recruits EVENT = 38 COST = penalty }
	}
	if = {
		limit = {
			has_modifier = nr_landowners_offices_chancery
			has_law = law_type:law_elected_bureaucrats
		}
		nr_campaign_interrupt = { KEY = nr_landowners_offices FORM = chancery EVENT = 39 COST = penalty }
	}
}

# ---------- Follow-ups: attach the amendment to the active law. Scope: country. ----------
nr_landowners_offices_add_patrimonial_justice = {
	if = {
		limit = { has_law = law_type:law_serfdom }
		nr_amendment_add_to_active_law = { AMENDMENT = amendment_nr_landowners_patrimonial_justice LAW = law_serfdom IG = ig_landowners }
	}
	else_if = {
		limit = { has_law = law_type:law_manorialism }
		nr_amendment_add_to_active_law = { AMENDMENT = amendment_nr_landowners_patrimonial_justice LAW = law_manorialism IG = ig_landowners }
	}
}
