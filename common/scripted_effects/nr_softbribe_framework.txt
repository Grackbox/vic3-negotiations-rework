# NR: "money or the group's own soft form" for negotiation option 1 (every group that has a soft form:
# Devout - charity; more to come). Decided once at the start of the negotiation, flag nr_soft_bribe on the group.
# Parameters: KEY = campaign slot key of the soft form (nr_campaign_slots), REFUSES = trigger (character scope):
# the leader never takes money. Scope: country, scope:ig = the group.
nr_softbribe_roll = {
	if = {
		# a bribe that would cause a default is never offered: the group asks for its own form instead
		limit = { nr_bribe_affordable = no }
		scope:ig = { set_variable = nr_soft_bribe }
	}
	else_if = {
		limit = { scope:ig = { leader ?= { $REFUSES$ = yes } } }
		scope:ig = { set_variable = nr_soft_bribe }
	}
	else_if = {
		# the group already runs as many campaigns of its form as its clout allows: only money is possible
		limit = { NOT = { nr_campaign_slot_free = { KEY = $KEY$ } } }
	}
	else_if = {
		limit = { scope:ig = { leader ?= { nr_bribe_leader_loves = yes } } }
		# money
	}
	else = {
		random_list = {
			0 = { # the group's own form, share nr_softbribe_share
				modifier = { value = scope:ig.nr_softbribe_share }
				scope:ig = { set_variable = nr_soft_bribe }
			}
			0 = { # money
				modifier = { value = scope:ig.nr_bribe_cash_share }
			}
		}
	}
}

# Sets the expenses of this campaign (its own slot modifier <KEY>_expenses_<slot>) to the base amount fixed at the
# negotiation (<KEY>_cost_amount) x $FACTOR$ (1 at the negotiation, the form factor later). The slot is
# <KEY>_current_slot (nr_campaign_take_slot). Guarded by has_variable: the form event can be inspected or fired by
# console without the variables. Scope: country.
nr_softbribe_set_cost = {
	if = {
		limit = {
			has_variable = $KEY$_cost_amount
			has_variable = $KEY$_current_slot
		}
		if = {
			limit = { var:$KEY$_current_slot = 1 }
			nr_softbribe_set_slot_cost = { KEY = $KEY$ SLOT = 1 FACTOR = $FACTOR$ }
		}
		else_if = {
			limit = { var:$KEY$_current_slot = 2 }
			nr_softbribe_set_slot_cost = { KEY = $KEY$ SLOT = 2 FACTOR = $FACTOR$ }
		}
		else = {
			nr_softbribe_set_slot_cost = { KEY = $KEY$ SLOT = 3 FACTOR = $FACTOR$ }
		}
	}
}

nr_softbribe_set_slot_cost = {
	if = {
		limit = { has_modifier = $KEY$_expenses_$SLOT$ }
		remove_modifier = $KEY$_expenses_$SLOT$
	}
	add_modifier = {
		name = $KEY$_expenses_$SLOT$
		multiplier = {
			value = var:$KEY$_cost_amount
			multiply = $FACTOR$
		}
		days = long_modifier_time
		is_decaying = yes
	}
}
