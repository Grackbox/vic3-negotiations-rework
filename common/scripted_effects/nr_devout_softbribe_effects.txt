# NR: Devout soft bribe - funds for the church's needs (negotiation option 1).
# The form of spending is chosen by event nr_devout_softbribe.1 a week later.

# Bribe or charity in this negotiation. Called from nr_negotiation_after_options.
nr_devout_softbribe_roll = {
	if = {
		# a bribe that would cause a default is never offered: the church asks for charity instead
		limit = { nr_devout_bribe_affordable = no }
		scope:ig = { set_variable = nr_soft_bribe }
	}
	else_if = {
		limit = { scope:ig = { leader ?= { nr_devout_softbribe_leader_refuses = yes } } }
		scope:ig = { set_variable = nr_soft_bribe }
	}
	else_if = {
		# the church already runs as many campaigns as its clout allows: only a regular bribe is possible
		limit = { NOT = { nr_campaign_slot_free = { KEY = nr_devout_softbribe } } }
	}
	else_if = {
		limit = { scope:ig = { leader ?= { nr_devout_softbribe_leader_loves = yes } } }
		# regular bribe
	}
	else = {
		random_list = {
			60 = { # chance of charity when the leader has no relevant traits
				scope:ig = { set_variable = nr_soft_bribe }
			}
			40 = {
				# regular bribe
			}
		}
	}
}

# Negotiation option. The base cost is shown right away; the campaign form is chosen by an event a week later.
nr_devout_softbribe_start = {
	custom_tooltip = nr_devout_softbribe_tt
	hidden_effect = {
		# Blocks a second Devout option 1 until the form event fires (see nr_neg_option_1_allowed)
		set_variable = {
			name = nr_devout_softbribe_pending
			days = 14
		}
		set_variable = {
			name = nr_devout_softbribe_cost_amount
			value = nr_devout_softbribe_base_cost
		}
		nr_campaign_take_slot = { KEY = nr_devout_softbribe }
	}
	add_modifier = {
		name = nr_devout_softbribe_base
		days = long_modifier_time
		is_decaying = yes
	}
	nr_devout_softbribe_set_cost = { FACTOR = 1 }
	scope:ig = {
		finish_negotiation = "negotiation.1.o1"
		add_modifier = {
			name = nr_devout_softbribe_patronage
			days = long_modifier_time
			multiplier = nr_neg_level_scale
			is_decaying = yes
		}
		improve_law_stance = yes
	}
	hidden_effect = {
		set_variable = {
			name = nr_devout_softbribe_degree
			value = scope:ig.amenability_level
		}
	}
	trigger_event = {
		id = nr_devout_softbribe.1
		days = 7 # delay of the campaign form event
		popup = yes
	}
}

# Offers one more random civil form in event nr_devout_softbribe.1 (sets nr_devout_softbribe_offer_<form>). Scope: country.
nr_devout_softbribe_offer_random_form = {
	random_list = {
		1 = {
			trigger = { nr_devout_softbribe_can_offer = { FORM = asceticism } }
			set_variable = nr_devout_softbribe_offer_asceticism
		}
		1 = {
			trigger = { nr_devout_softbribe_can_offer = { FORM = enlightenment } }
			set_variable = nr_devout_softbribe_offer_enlightenment
		}
		1 = {
			trigger = { nr_devout_softbribe_can_offer = { FORM = sisters } }
			set_variable = nr_devout_softbribe_offer_sisters
		}
		1 = {
			trigger = { nr_devout_softbribe_can_offer = { FORM = shelters } }
			set_variable = nr_devout_softbribe_offer_shelters
		}
	}
}

# Form modifier for 10 years, decaying. Scope: country. $FORM$ = asceticism / enlightenment / ...
nr_devout_softbribe_add_form = {
	add_modifier = {
		name = nr_devout_softbribe_$FORM$
		days = long_modifier_time
		is_decaying = yes
	}
}

# Sets the expenses of this campaign (its own slot modifier nr_devout_softbribe_expenses_<slot>) to the base amount
# fixed at the negotiation x $FACTOR$ (1 at the negotiation; 0.25 / 0.5 / 1.25 / 1.5 for the form).
# Guarded by has_variable: the event can be inspected or fired by console without the variables. Scope: country.
nr_devout_softbribe_set_cost = {
	if = {
		limit = {
			has_variable = nr_devout_softbribe_cost_amount
			has_variable = nr_devout_softbribe_current_slot
		}
		if = {
			limit = { var:nr_devout_softbribe_current_slot = 1 }
			nr_devout_softbribe_set_slot_cost = { SLOT = 1 FACTOR = $FACTOR$ }
		}
		else_if = {
			limit = { var:nr_devout_softbribe_current_slot = 2 }
			nr_devout_softbribe_set_slot_cost = { SLOT = 2 FACTOR = $FACTOR$ }
		}
		else = {
			nr_devout_softbribe_set_slot_cost = { SLOT = 3 FACTOR = $FACTOR$ }
		}
	}
}

nr_devout_softbribe_set_slot_cost = {
	if = {
		limit = { has_modifier = nr_devout_softbribe_expenses_$SLOT$ }
		remove_modifier = nr_devout_softbribe_expenses_$SLOT$
	}
	add_modifier = {
		name = nr_devout_softbribe_expenses_$SLOT$
		multiplier = {
			value = var:nr_devout_softbribe_cost_amount
			multiply = $FACTOR$
		}
		days = long_modifier_time
		is_decaying = yes
	}
}

# Clears the data of event nr_devout_softbribe.1. Scope: country.
nr_devout_softbribe_cleanup = {
	remove_variable = nr_devout_softbribe_pending
	remove_variable = nr_devout_softbribe_cost_amount
	remove_variable = nr_devout_softbribe_degree
	remove_variable = nr_devout_softbribe_current_slot
	remove_variable = nr_devout_softbribe_offer_asceticism
	remove_variable = nr_devout_softbribe_offer_enlightenment
	remove_variable = nr_devout_softbribe_offer_sisters
	remove_variable = nr_devout_softbribe_offer_shelters
}

# Colonial missions: one year of conversion in every colonial subject state. Scope: overlord country.
nr_devout_softbribe_mission_convert = {
	religion = { save_scope_as = nr_devout_softbribe_overlord_religion }
	every_subject_or_below = {
		limit = { nr_devout_softbribe_is_mission_subject = yes }
		every_scope_state = {
			convert_population = {
				target = scope:nr_devout_softbribe_overlord_religion
				value = nr_devout_softbribe_mission_share
			}
		}
	}
}

# Colonial missions: start. Converts right away, then yearly via hidden event nr_devout_softbribe.2.
nr_devout_softbribe_mission_start = {
	custom_tooltip = nr_devout_softbribe_missions_tt
	every_subject_or_below = {
		limit = {
			nr_devout_softbribe_is_mission_subject = yes
			NOT = { religion = root.religion } # same state religion: converting its pops is no imposition
		}
		add_modifier = {
			name = nr_devout_softbribe_mission_pressure
			days = long_modifier_time
			is_decaying = yes
		}
	}
	hidden_effect = {
		nr_devout_softbribe_mission_convert = yes
		set_variable = {
			name = nr_devout_softbribe_mission_years_left
			value = {
				value = nr_devout_softbribe_mission_years
				subtract = 1
			}
		}
		trigger_event = {
			id = nr_devout_softbribe.2
			days = 365
		}
	}
}
