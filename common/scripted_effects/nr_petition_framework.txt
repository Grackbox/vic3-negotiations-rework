# NR: reusable helpers for the "IG petition" pattern: a journal entry asks for a law for 4 years; while that law
# is being enacted it gets an enactment speed bonus once per attempt; timeout without the law upsets the group.
# Parameters: PETITION = key prefix (modifiers <PETITION>_speed and <PETITION>_ignored, variable <PETITION>_boosted),
# LAW = law key. Scope: country; the journal entry saves the group as scope:ig.

# Gives the enactment of the law the speed bonus once per enactment attempt. Called on creation and monthly.
nr_petition_boost = {
	if = {
		limit = {
			is_enacting_law = law_type:$LAW$
			NOT = { has_variable = $PETITION$_boosted }
		}
		add_enactment_modifier = {
			name = $PETITION$_speed
		}
		set_variable = $PETITION$_boosted
	}
	else_if = {
		limit = {
			NOT = { is_enacting_law = law_type:$LAW$ }
			has_variable = $PETITION$_boosted
		}
		remove_variable = $PETITION$_boosted # enactment ended or failed: boost the next attempt again
	}
}

# The petition is over (law passed or timed out).
nr_petition_end = {
	hidden_effect = { remove_variable = $PETITION$_boosted }
}

# Timeout without the law: the group is upset for 5 years.
nr_petition_timeout = {
	scope:ig ?= {
		add_modifier = {
			name = $PETITION$_ignored
			days = normal_modifier_time
		}
	}
	nr_petition_end = { PETITION = $PETITION$ }
}

# Called from nr_on_law_enactment_started: a group with an active petition for the law being enacted supports it
# (improve_stance = 1: neutral becomes approve, only for this enactment). Scope: country.
nr_petition_improve_stance = {
	if = {
		limit = { nr_devout_petition_for_enacting_law = yes }
		ig:ig_devout ?= { improve_stance = 1 }
	}
}

# Called from nr_on_law_enactment_started: every running Devout petition applies its enactment speed bonus at once,
# instead of waiting for the monthly pulse of its journal entry. Add every new Devout petition here. Scope: country.
nr_devout_petition_boost_all = {
	if = {
		limit = { has_journal_entry = je_nr_devout_schools_petition }
		nr_petition_boost = { PETITION = nr_devout_schools_petition LAW = law_religious_schools }
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_sisters_petition }
		nr_petition_boost = { PETITION = nr_devout_sisters_petition LAW = law_charitable_health_system }
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_censorship_petition }
		nr_petition_boost = { PETITION = nr_devout_censorship_petition LAW = law_censorship }
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_sunday_petition }
		nr_petition_boost = { PETITION = nr_devout_sunday_petition LAW = law_regulatory_bodies }
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_census_petition }
		nr_petition_boost = { PETITION = nr_devout_census_petition LAW = law_appointed_bureaucrats }
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_police_petition }
		nr_devout_police_petition_boost = yes
	}
	if = {
		limit = { has_journal_entry = je_nr_devout_slavery_petition }
		nr_petition_boost = { PETITION = nr_devout_slavery_petition LAW = law_slavery_banned }
	}
}
