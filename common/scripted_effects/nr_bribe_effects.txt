# NR: regular bribe (negotiation option 1). Called from negotiation.1 option negotiation.1.o1. Scope: country, scope:ig set.
# Every group: one-time payment instead of the vanilla weekly expenses, refund of the unspent part when exposed.
# Devout: also a trait roll for the leader who took the money.

# Payment
nr_bribe_pay = {
	# stored on the group for the refund in generic_laws.2 (a second bribed group does not overwrite it)
	scope:ig = {
		set_variable = {
			name = nr_bribe_paid
			value = root.nr_bribe_lump_sum
			days = 330
		}
	}
	# computed directly, not from the variable: tooltips are built before the variable is set
	add_treasury = {
		value = nr_bribe_lump_sum
		multiply = -1
	}
}

# Exposure roll: same chances and delay as vanilla (10%, 20% under Protected Speech; generic_laws.2 in 300 days).
# Devout: the leader who took the money is marked; 310 days later nr_bribe.1 rolls a trait if the bribe was not exposed.
nr_bribe_exposure_roll = {
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		scope:ig = {
			leader ?= {
				set_variable = {
					name = nr_bribe_taken
					days = 330
				}
			}
		}
		custom_tooltip = nr_bribe_vice_risk_tt
		hidden_effect = { # hidden event: a blank "an event fires" line would show otherwise
			trigger_event = {
				id = nr_bribe.1
				days = 310
			}
		}
	}
	if = {
		limit = { has_law = law_type:law_protected_speech }
		random = {
			chance = 20
			nr_bribe_expose = yes
		}
	}
	else = {
		random = {
			chance = 10
			nr_bribe_expose = yes
		}
	}
}

nr_bribe_expose = {
	trigger_event = {
		id = generic_laws.2
		popup = yes
		days = 300
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		scope:ig = {
			leader ?= {
				set_variable = {
					name = nr_bribe_exposed
					days = 330
				}
			}
		}
	}
}

# Refund of the unspent bribe money when the bribe is exposed. Scope: country; scope:corrupt_ig_scope from generic_laws.2.
nr_bribe_refund = {
	if = {
		limit = {
			scope:corrupt_ig_scope ?= { has_variable = nr_bribe_paid }
		}
		add_treasury = scope:corrupt_ig_scope.nr_bribe_refund
		scope:corrupt_ig_scope = { remove_variable = nr_bribe_paid }
	}
}

# Trait for a character who kept the money quietly. Scope: character. Nothing if he already has all of them.
nr_bribe_add_vice_trait = {
	random_list = {
		25 = {
			trigger = { NOT = { has_trait = grifter } }
			add_trait = grifter
		}
		25 = {
			trigger = { NOT = { has_trait = expensive_tastes } }
			add_trait = expensive_tastes
		}
		20 = {
			trigger = { NOT = { has_trait = alcoholic } }
			add_trait = alcoholic
		}
		15 = {
			trigger = { NOT = { has_trait = opium_addiction } }
			add_trait = opium_addiction
		}
		15 = {
			trigger = { NOT = { has_trait = syphilis } }
			add_trait = syphilis
		}
	}
}
