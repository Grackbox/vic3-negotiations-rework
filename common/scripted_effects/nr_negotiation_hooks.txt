# NR: effects called by the hooks in vanilla negotiations. Only routing by interest group here;
# the tasks themselves live in nr_<group>_<task>_*.txt files.

# Called in the immediate of negotiation.1 after set_neg_options: overrides precomputed option values.
nr_negotiation_after_options = {
	# Option 1: bribe or soft bribe - decided once at the start of the negotiation
	scope:ig = {
		if = {
			limit = { has_variable = nr_soft_bribe }
			remove_variable = nr_soft_bribe
		}
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		nr_softbribe_roll = { KEY = nr_devout_softbribe REFUSES = nr_devout_softbribe_leader_refuses }
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		nr_softbribe_roll = { KEY = nr_landowners_grants REFUSES = nr_landowners_grants_leader_refuses }
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		nr_softbribe_roll = { KEY = nr_industrialists_contracts REFUSES = nr_industrialists_contracts_leader_refuses }
		nr_industrialists_healthy_prepare = yes # heavy industry promises of options 5 and 6
	}
	if = {
		limit = {
			scope:ig = {
				is_interest_group_type = ig_devout
				has_variable = promised_sol_level
			}
		}
		nr_devout_sol_set_target = yes
	}
	if = {
		limit = {
			scope:ig = {
				is_interest_group_type = ig_landowners
				has_variable = promised_sol_level
			}
		}
		nr_landowners_sol_set_target = yes
	}
	if = {
		limit = {
			scope:ig = {
				is_interest_group_type = ig_industrialists
				has_variable = promised_sol_level
			}
		}
		nr_industrialists_sol_set_target = yes
	}
}

# Option 9 (SoL) for groups in nr_neg_option_9_is_custom.
nr_neg_option_9_custom = {
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		nr_devout_sol_promise = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		nr_landowners_sol_promise = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		nr_industrialists_sol_promise = yes
	}
}

# Option 1: soft bribe for groups in nr_neg_option_1_is_custom.
nr_neg_option_1_custom = {
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		nr_devout_softbribe_start = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		nr_landowners_grants_start = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		nr_industrialists_contracts_start = yes
	}
}

# Option 2: the group's own posts for groups in nr_neg_option_2_is_custom.
nr_neg_option_2_custom = {
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		nr_devout_officials_start = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		nr_landowners_offices_start = yes
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		nr_industrialists_commissions_start = yes
	}
}

# Called in negotiation.1 right after the vanilla set_promised_building_type (option 5). Scope: interest group.
nr_after_promised_building_type = {
	if = {
		limit = { is_interest_group_type = ig_devout }
		nr_devout_fish_roll = yes
	}
}

# Option 5 (buildings): lore line for the building requests added by the mod. Scope: country, scope:ig.
nr_neg_option_5_flavor = {
	if = {
		limit = {
			scope:ig = {
				is_interest_group_type = ig_devout
				var:promised_building_type ?= bt:building_fishing_wharf
			}
		}
		custom_tooltip = nr_devout_fish_lore
	}
}
