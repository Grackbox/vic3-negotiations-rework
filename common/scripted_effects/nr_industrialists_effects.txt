# NR: the Industrialists' state contracts (option 1, instead of money) and places in the commissions (option 2).
# Same schemes as the landowners' grants and places in the provinces.

# ========== State contracts ==========
# Negotiation option. Scope: country, scope:ig = Industrialists.
nr_industrialists_contracts_start = {
	custom_tooltip = nr_industrialists_contracts_tt
	hidden_effect = {
		set_variable = {
			name = nr_industrialists_contracts_pending
			days = 14
		}
		set_variable = {
			name = nr_industrialists_contracts_cost_amount
			value = neg_bribe_amount
		}
		set_variable = {
			name = nr_industrialists_contracts_degree
			value = scope:ig.amenability_level
		}
		nr_campaign_take_slot = { KEY = nr_industrialists_contracts }
	}
	custom_tooltip = nr_softbribe_cost_tt # the expenses modifier is set from hidden variables, so the tooltip would not show it
	nr_softbribe_set_cost = { KEY = nr_industrialists_contracts FACTOR = 1 }
	scope:ig = {
		finish_negotiation = "negotiation.1.o1"
		add_modifier = {
			name = nr_industrialists_contracts_favor
			days = long_modifier_time
			multiplier = nr_neg_level_scale
			is_decaying = yes
		}
		improve_law_stance = yes
	}
	trigger_event = {
		id = nr_industrialists_contracts.1
		days = 7
		popup = yes
	}
}

# Chosen form: its modifier for 10 years, its expenses (set by nr_softbribe_set_cost in the option), the slot it occupies, the running mark for its follow-up. Scope: country.
nr_industrialists_contracts_choose_form = {
	add_modifier = {
		name = nr_industrialists_contracts_$FORM$
		days = long_modifier_time
		is_decaying = yes
	}
	hidden_effect = {
		if = {
			limit = { has_variable = nr_industrialists_contracts_current_slot }
			set_variable = {
				name = nr_industrialists_contracts_$FORM$_slot
				value = var:nr_industrialists_contracts_current_slot
				days = long_modifier_time
			}
		}
		set_variable = {
			name = nr_industrialists_contracts_$FORM$_running
			days = 3660
		}
	}
}

# Railway Concession: new railways (level 1) in up to 2 incorporated states without one, the most populous first.
nr_industrialists_contracts_build_railways = {
	ordered_scope_state = {
		limit = {
			is_incorporated = yes
			NOT = { has_building = building_railway }
		}
		order_by = state_population
		position = 0
		save_scope_as = nr_railway_state_1
		custom_tooltip = nr_industrialists_contracts_railway_built_1_tt # the vanilla Russian tooltip of create_building is broken
		hidden_effect = {
			create_building = {
				building = building_railway
				level = 1
			}
		}
	}
	ordered_scope_state = {
		limit = {
			is_incorporated = yes
			NOT = { has_building = building_railway }
			NOT = { this = scope:nr_railway_state_1 } # the first line may not be registered yet
		}
		order_by = state_population
		position = 0
		save_scope_as = nr_railway_state_2
		custom_tooltip = nr_industrialists_contracts_railway_built_2_tt
		hidden_effect = {
			create_building = {
				building = building_railway
				level = 1
			}
		}
	}
}

# A law change ends a form (called from nr_on_law_activated). Scope: country.
nr_industrialists_contracts_check_laws = {
	if = {
		limit = {
			has_modifier = nr_industrialists_contracts_iron_tariff
			has_law = law_type:law_free_trade
		}
		nr_industrialists_contracts_interrupt = { FORM = iron_tariff EVENT = 21 }
	}
	if = {
		limit = {
			has_modifier = nr_industrialists_contracts_charter
			has_law = law_type:law_no_colonial_affairs
		}
		nr_industrialists_contracts_interrupt = { FORM = charter EVENT = 22 }
	}
	if = {
		limit = {
			has_modifier = nr_industrialists_contracts_strikes
			OR = { has_law = law_type:law_right_to_associate has_law = law_type:law_corporatized_unions has_law = law_type:law_factory_councils }
		}
		nr_industrialists_contracts_interrupt = { FORM = strikes EVENT = 23 }
	}
	if = {
		limit = {
			has_modifier = nr_industrialists_contracts_contract_labour
			NOT = { has_law = law_type:law_no_migration_controls }
		}
		nr_industrialists_contracts_interrupt = { FORM = contract_labour EVENT = 24 }
	}
}

# Ends a form early: its modifier, its costs and its slot go, no follow-up; popup $EVENT$ (pattern "Campaign interruption").
nr_industrialists_contracts_interrupt = {
	remove_modifier = nr_industrialists_contracts_$FORM$
	hidden_effect = {
		if = {
			limit = { var:nr_industrialists_contracts_$FORM$_slot ?= 1 }
			nr_industrialists_contracts_free_slot = { SLOT = 1 }
		}
		else_if = {
			limit = { var:nr_industrialists_contracts_$FORM$_slot ?= 2 }
			nr_industrialists_contracts_free_slot = { SLOT = 2 }
		}
		else_if = {
			limit = { var:nr_industrialists_contracts_$FORM$_slot ?= 3 }
			nr_industrialists_contracts_free_slot = { SLOT = 3 }
		}
		if = {
			limit = { has_variable = nr_industrialists_contracts_$FORM$_slot }
			remove_variable = nr_industrialists_contracts_$FORM$_slot
		}
		if = {
			limit = { has_variable = nr_industrialists_contracts_$FORM$_running }
			remove_variable = nr_industrialists_contracts_$FORM$_running
		}
	}
	trigger_event = { id = nr_industrialists_contracts.$EVENT$ popup = yes }
}

nr_industrialists_contracts_free_slot = {
	if = {
		limit = { has_modifier = nr_industrialists_contracts_expenses_$SLOT$ }
		remove_modifier = nr_industrialists_contracts_expenses_$SLOT$
	}
	if = {
		limit = { has_variable = nr_industrialists_contracts_slot_$SLOT$ }
		remove_variable = nr_industrialists_contracts_slot_$SLOT$
	}
}

nr_industrialists_commissions_set_penalty = {
	if = {
		limit = { has_variable = nr_industrialists_commissions_current_slot }
		if = {
			limit = { var:nr_industrialists_commissions_current_slot = 1 }
			nr_industrialists_commissions_set_slot_penalty = { SLOT = 1 FACTOR = $FACTOR$ }
		}
		else_if = {
			limit = { var:nr_industrialists_commissions_current_slot = 2 }
			nr_industrialists_commissions_set_slot_penalty = { SLOT = 2 FACTOR = $FACTOR$ }
		}
		else = {
			nr_industrialists_commissions_set_slot_penalty = { SLOT = 3 FACTOR = $FACTOR$ }
		}
	}
}

nr_industrialists_commissions_set_slot_penalty = {
	if = {
		limit = { has_modifier = nr_industrialists_commissions_penalty_$SLOT$ }
		remove_modifier = nr_industrialists_commissions_penalty_$SLOT$
	}
	add_modifier = {
		name = nr_industrialists_commissions_penalty_$SLOT$
		multiplier = {
			value = nr_industrialists_commissions_level
			multiply = $FACTOR$
		}
		days = long_modifier_time
		is_decaying = yes
	}
}

# ========== Heavy industry promise (options 5 and 6) ==========
# Called from nr_negotiation_after_options (after set_neg_options): prepares the targets for the option tooltips.
# Option 5: the vanilla type if it is heavy industry, target = vanilla building_level_increase (existing + N).
# Option 6: a heavy industry type the country already has instead of the whole group, target = its levels + N,
# N = the vanilla group increase (with our logarithmic scaler). Scope: country, scope:ig.
nr_industrialists_healthy_prepare = {
	scope:ig = {
		if = {
			limit = { has_variable = nr_industrialists_healthy_type_5 }
			remove_variable = nr_industrialists_healthy_type_5
		}
		if = {
			limit = { has_variable = nr_industrialists_healthy_type_6 }
			remove_variable = nr_industrialists_healthy_type_6
		}
		if = {
			limit = {
				exists = var:promised_building_type
				var:promised_building_type = { nr_industrialists_healthy_is_heavy = yes }
				exists = var:building_level_increase
			}
			set_variable = {
				name = nr_industrialists_healthy_type_5
				value = var:promised_building_type
			}
			set_variable = {
				name = nr_industrialists_healthy_target_5
				value = var:building_level_increase
			}
		}
		if = {
			limit = {
				var:promised_building_group ?= bg:bg_heavy_industry
				exists = var:building_group_level_increase
				owner = {
					any_scope_building = {
						is_building_group = bg_heavy_industry
						type = { nr_industrialists_healthy_is_heavy = yes }
					}
				}
			}
			owner = {
				random_scope_building = {
					limit = {
						is_building_group = bg_heavy_industry
						type = { nr_industrialists_healthy_is_heavy = yes }
					}
					type = { save_temporary_scope_as = nr_industrialists_healthy_group_type }
				}
			}
			set_variable = {
				name = nr_industrialists_healthy_type_6
				value = scope:nr_industrialists_healthy_group_type
			}
			set_variable = {
				name = nr_industrialists_healthy_target_6
				value = {
					value = nr_industrialists_healthy_all_levels_6
					add = var:building_group_level_increase
					subtract = building_group_levels_existing
					min = {
						value = nr_industrialists_healthy_all_levels_6
						add = 1
					}
				}
			}
		}
	}
}
