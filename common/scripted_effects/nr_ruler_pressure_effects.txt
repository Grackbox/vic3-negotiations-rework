# NR: the ruler presses the group in negotiations (feature ruler_pressure, tables in README_NR.md).
# Called from option negotiation.1.nr_pressure. Scope: country, scope:ig, scope:law.

nr_ruler_pressure_apply = {
	nr_ruler_pressure_lore = yes
	nr_authority_cost_add = { COST = 400 }
	if = { # a monarch or theocrat with an elected parliament meddles in its affairs
		limit = { nr_ruler_pressure_constitutional = yes }
		add_modifier = { name = nr_ruler_pressure_constitutional days = normal_modifier_time }
	}
	if = {
		limit = { nr_ruler_pressure_own_group = yes }
		nr_ruler_pressure_own = yes
	}
	else = {
		nr_ruler_pressure_other = yes
	}
	add_to_variable_list = { # the 5-year limit is per group
		name = nr_ruler_pressure_pressed
		target = scope:ig
		years = 5
	}
	scope:ig = {
		finish_negotiation = "negotiation.1.nr_pressure"
		improve_law_stance = yes
	}
}

# Lore line by ruler type.
nr_ruler_pressure_lore = {
	if = {
		limit = { nr_ruler_is_monarch = yes }
		custom_tooltip = nr_ruler_pressure_lore_monarch
	}
	else_if = {
		limit = { nr_ruler_is_theocrat = yes }
		custom_tooltip = nr_ruler_pressure_lore_theocrat
	}
	else_if = {
		limit = { nr_ruler_is_autocrat = yes }
		custom_tooltip = nr_ruler_pressure_lore_autocrat
	}
	else_if = {
		limit = { nr_ruler_is_oligarchs = yes }
		custom_tooltip = nr_ruler_pressure_lore_oligarchs
	}
	else_if = {
		limit = { nr_ruler_is_technocrats = yes }
		custom_tooltip = nr_ruler_pressure_lore_technocrats
	}
}

# Armed Forces pressed by a ruler from another group: by the loyalty of their leader.
nr_ruler_pressure_army = {
	if = {
		limit = { nr_ruler_pressure_army_loyal = yes }
		if = {
			limit = { nr_ruler_is_monarch = yes }
			add_modifier = { name = nr_ruler_pressure_army_loyal_monarch days = normal_modifier_time }
		}
		else_if = {
			limit = { nr_ruler_is_autocrat = yes }
			add_modifier = { name = nr_ruler_pressure_army_loyal_autocrat days = normal_modifier_time }
		}
		else = {
			custom_tooltip = nr_ruler_pressure_order_obeyed_tt
		}
	}
	else_if = {
		limit = { nr_ruler_pressure_army_wary = yes }
		add_modifier = { name = nr_ruler_pressure_army_wary days = normal_modifier_time multiplier = nr_ruler_pressure_scale }
	}
	else = {
		add_modifier = { name = nr_ruler_pressure_army_disloyal days = normal_modifier_time multiplier = nr_ruler_pressure_scale }
		scope:ig.leader ?= {
			add_modifier = { name = nr_ruler_pressure_leader_disloyal days = normal_modifier_time }
		}
	}
}
