# NR: Devout schools (feature devout_schools).

# Gives the Religious Schools enactment +25% speed once per enactment. Scope: country.
nr_devout_schools_petition_boost = {
	if = {
		limit = {
			is_enacting_law = law_type:law_religious_schools
			NOT = { has_variable = nr_devout_schools_petition_boosted }
		}
		add_enactment_modifier = {
			name = nr_devout_schools_petition_speed
		}
		set_variable = nr_devout_schools_petition_boosted
	}
	else_if = {
		limit = {
			NOT = { is_enacting_law = law_type:law_religious_schools }
			has_variable = nr_devout_schools_petition_boosted
		}
		remove_variable = nr_devout_schools_petition_boosted # enactment ended or failed: boost the next attempt again
	}
}
