# NR: regular bribe (negotiation option 1).

# The bribe can be paid without reaching the credit limit. Scope: country, scope:ig = the bribed group.
nr_bribe_affordable = {
	nr_bribe_room_after >= 0
}

# AI (option negotiation.1.o1, via nr_neg_option_1_ai_*).
# Paid from gold reserves, no new debt.
nr_bribe_ai_good = {
	nr_bribe_reserves_after >= 0
}

# Partly on credit, but at least half of the credit stays free.
nr_bribe_ai_fair = {
	nr_bribe_reserves_after < 0
	nr_bribe_room_after_share >= 0.5
}

# On credit while in deficit, or less than a quarter of the credit left.
nr_bribe_ai_bad = {
	nr_bribe_reserves_after < 0
	OR = {
		net_total_income < 0
		nr_bribe_room_after_share < 0.25
	}
}

# ---------- Leader traits (every group). Scope: character. ----------
# Loves money: always takes a regular bribe, +10 to the weight of option 1.
nr_bribe_leader_loves = {
	OR = {
		has_trait = grifter
		has_trait = expensive_tastes
		has_trait = hedonist
	}
}

# Will not take money: always the group's own soft form, -10 to the weight of option 1 (groups other than the Devout,
# whose leader weight and refusal traits are their own: nr_devout_softbribe_leader_weight / _refuses).
nr_bribe_leader_refuses = {
	has_trait = honorable
}

# Option 1 can be rolled for a group that has its own soft form ($KEY$ = campaign slot key, $REFUSES$ = trigger:
# the leader never takes money): no second option 1 while the previous form event is pending ($KEY$_pending),
# and either a free slot for the form or a regular bribe the leader would take. Scope: country, scope:ig.
nr_softbribe_option_allowed = {
	NOT = { has_variable = $KEY$_pending }
	OR = {
		nr_campaign_slot_free = { KEY = $KEY$ }
		AND = {
			nr_bribe_affordable = yes
			NOT = { scope:ig.leader ?= { $REFUSES$ = yes } }
		}
	}
}
