# NR: the Industrialists' features (state contracts, commissions). Scope: country unless noted.

# The Industrialists leader is not against the law (without scope:ig).
nr_industrialists_leader_not_against_law = {
	NOT = {
		ig:ig_industrialists ?= {
			leader ?= {
				law_stance = {
					law = law_type:$LAW$
					value < neutral
				}
			}
		}
	}
}

# The Industrialists leader never takes money for a contract: Honorable, or Ambitious (an empire, not an envelope). Scope: character.
nr_industrialists_contracts_leader_refuses = {
	OR = {
		nr_bribe_leader_refuses = yes
		has_trait = ambitious
	}
}

# An Industrialists' petition asks for the law being enacted.
nr_industrialists_petition_for_enacting_law = {
	OR = {
		AND = {
			has_journal_entry = je_nr_industrialists_protectionism_petition
			is_enacting_law = law_type:law_protectionism
		}
		AND = {
			has_journal_entry = je_nr_industrialists_colonial_petition
			is_enacting_law = law_type:law_colonial_exploitation
		}
		AND = {
			has_journal_entry = je_nr_industrialists_combination_petition
			is_enacting_law = law_type:law_combination_acts
		}
	}
}

	OR = {
		this = bt:building_steel_mill
		this = bt:building_chemical_plant
		this = bt:building_explosives_factory
		this = bt:building_synthetics_plant
		this = bt:building_motor_industry
		this = bt:building_automotive_industry
		this = bt:building_electrics_industry
	}
}
