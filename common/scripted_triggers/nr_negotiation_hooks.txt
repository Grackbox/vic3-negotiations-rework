# NR: triggers used by the hooks in vanilla negotiations.
# New groups and tasks are added here; the vanilla copies stay untouched.

# Option 9 (SoL): does this group use our commitment instead of vanilla je_negotiate_sol?
nr_neg_option_9_is_custom = {
	scope:ig = {
		OR = {
			is_interest_group_type = ig_devout
		}
	}
}

# Is any of our SoL commitments active (at most one at a time, like vanilla)
nr_has_custom_sol_promise = {
	OR = {
		has_journal_entry = je_nr_devout_sol
	}
}

# ---------- Option 1 (bribe) ----------
# Soft bribe instead of a bribe in this negotiation? Decided in nr_negotiation_after_options.
nr_neg_option_1_is_custom = {
	scope:ig = {
		has_variable = nr_soft_bribe
	}
}

# Can option 1 be rolled at all. Scope: country.
nr_neg_option_1_allowed = {
	# Devout: no second option 1 while the previous soft bribe waits for its form event
	OR = {
		scope:ig = { NOT = { is_interest_group_type = ig_devout } }
		NOT = { has_variable = nr_devout_softbribe_pending }
	}
}

# Can the regular bribe (negotiation.1.o1) still be chosen. Scope: country.
nr_neg_option_1_affordable = {
	# Devout: a bribe that would cause a default is not offered
	OR = {
		scope:ig = { NOT = { is_interest_group_type = ig_devout } }
		nr_devout_bribe_affordable = yes
	}
}

# AI weights of the regular bribe. Vanilla income checks apply to groups that pay the vanilla weekly bribe;
# groups with their own payment get the good / fair / bad weights instead (+35 / +15 / -50).
nr_neg_option_1_vanilla_ai = {
	scope:ig = { NOT = { is_interest_group_type = ig_devout } }
}

nr_neg_option_1_ai_good = {
	scope:ig = { is_interest_group_type = ig_devout }
	nr_devout_bribe_ai_good = yes
}

nr_neg_option_1_ai_fair = {
	scope:ig = { is_interest_group_type = ig_devout }
	nr_devout_bribe_ai_fair = yes
}

nr_neg_option_1_ai_bad = {
	scope:ig = { is_interest_group_type = ig_devout }
	nr_devout_bribe_ai_bad = yes
}

# ---------- Option 2 (places in the administration) ----------
# Devout get their own version of option 2 (church officials, nr_devout_officials_*).
nr_neg_option_2_is_custom = {
	scope:ig = { is_interest_group_type = ig_devout }
}

# Can option 2 be rolled at all. Scope: country. Vanilla already blocks it while negotiation_bureaucracy is active.
# No extra conditions for now (a possible bureaucracy deficit is the player's choice); kept as the hook point.
nr_neg_option_2_allowed = {
	always = yes
}

# AI weights of the custom option 2 (+25 / -30).
nr_neg_option_2_ai_good = {
	scope:ig = { is_interest_group_type = ig_devout }
	nr_devout_officials_ai_good = yes
}

nr_neg_option_2_ai_bad = {
	scope:ig = { is_interest_group_type = ig_devout }
	nr_devout_officials_ai_bad = yes
}
