# NR: triggers used by the hooks in vanilla negotiations.
# New groups and tasks are added here; the vanilla copies stay untouched.

# Option 9 (SoL): does this group use our commitment instead of vanilla je_negotiate_sol?
nr_neg_option_9_is_custom = {
	scope:ig = {
		OR = {
			is_interest_group_type = ig_devout
		}
	}
}

# Is any of our SoL commitments active (at most one at a time, like vanilla)
nr_has_custom_sol_promise = {
	OR = {
		has_journal_entry = je_nr_devout_sol
	}
}

# ---------- Option 1 (bribe) ----------
# Soft bribe instead of a bribe in this negotiation? Decided in nr_negotiation_after_options.
nr_neg_option_1_is_custom = {
	scope:ig = {
		has_variable = nr_soft_bribe
	}
}

# Can option 1 be rolled at all. Scope: country.
nr_neg_option_1_allowed = {
	OR = {
		AND = {
			# groups without their own form yet: only a bribe the country can pay without a default
			scope:ig = {
				NOR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_landowners
					is_interest_group_type = ig_industrialists
				}
			}
			nr_bribe_affordable = yes
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_devout }
			nr_softbribe_option_allowed = { KEY = nr_devout_softbribe REFUSES = nr_devout_softbribe_leader_refuses }
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_landowners }
			nr_softbribe_option_allowed = { KEY = nr_landowners_grants REFUSES = nr_landowners_grants_leader_refuses }
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_industrialists }
			nr_softbribe_option_allowed = { KEY = nr_industrialists_contracts REFUSES = nr_industrialists_contracts_leader_refuses }
		}
	}
}

# Can the regular bribe (negotiation.1.o1) still be chosen. Scope: country.
# Every group is paid once: a bribe that would cause a default is not offered.
nr_neg_option_1_affordable = {
	nr_bribe_affordable = yes
}

# AI weights of the regular bribe. The vanilla income checks were written for the weekly bribe, which no group gets
# any more; every group uses the good / fair / bad weights of the one-time payment instead (+35 / +15 / -50).
nr_neg_option_1_vanilla_ai = {
	always = no
}

nr_neg_option_1_ai_good = {
	nr_bribe_ai_good = yes
}

nr_neg_option_1_ai_fair = {
	nr_bribe_ai_fair = yes
}

nr_neg_option_1_ai_bad = {
	nr_bribe_ai_bad = yes
}

# ---------- Option 2 (places in the administration) ----------
# Groups with their own version of option 2: Devout (church officials, nr_devout_officials_*),
# Landowners (places in the provinces, nr_landowners_offices_*).
nr_neg_option_2_is_custom = {
	scope:ig = {
		OR = {
			is_interest_group_type = ig_devout
			is_interest_group_type = ig_landowners
			is_interest_group_type = ig_industrialists
		}
	}
}

# Can option 2 be rolled at all. Scope: country.
# A possible bureaucracy deficit is the player's choice and is not checked.
nr_neg_option_2_allowed = {
	# Devout: church officials only while the church runs fewer campaigns than its clout allows
	OR = {
		scope:ig = {
			NOR = {
				is_interest_group_type = ig_devout
				is_interest_group_type = ig_landowners
				is_interest_group_type = ig_industrialists
			}
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_industrialists }
			nr_campaign_slot_free = { KEY = nr_industrialists_commissions }
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_devout }
			nr_campaign_slot_free = { KEY = nr_devout_officials }
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_landowners }
			nr_campaign_slot_free = { KEY = nr_landowners_offices }
		}
	}
}

# Vanilla allows one "places in the administration" deal at a time (negotiation_bureaucracy). Groups with their own
# option 2 use their own penalty modifiers and campaign limit, so they skip that check.
nr_neg_option_2_skips_vanilla_block = {
	nr_neg_option_2_is_custom = yes
}

# AI weights of the custom option 2 (+25 / -30).
nr_neg_option_2_ai_good = {
	OR = {
		AND = {
			scope:ig = { is_interest_group_type = ig_devout }
			nr_devout_officials_ai_good = yes
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_landowners }
			nr_landowners_offices_ai_good = yes
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_industrialists }
			nr_industrialists_commissions_ai_good = yes
		}
	}
}

nr_neg_option_2_ai_bad = {
	OR = {
		AND = {
			scope:ig = { is_interest_group_type = ig_devout }
			nr_devout_officials_ai_bad = yes
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_landowners }
			nr_landowners_offices_ai_bad = yes
		}
		AND = {
			scope:ig = { is_interest_group_type = ig_industrialists }
			nr_industrialists_commissions_ai_bad = yes
		}
	}
}
