# There is room on the law for a new amendment sponsored by $IG$. Scope: law.
nr_amendment_fits = {
	NOT = {
		any_scope_amendment = {
			count >= 2
			nr_amendment_is_ours = yes
		}
	}
	nr_amendment_opposed_$IG$ = no
}

# The law carries an amendment of a group opposing ig_armed_forces on this law. Scope: law.
nr_amendment_opposed_ig_armed_forces = {
	always = no # no opposing group on any law yet
}

# The law carries an amendment of a group opposing ig_devout on this law. Scope: law.
nr_amendment_opposed_ig_devout = {
	OR = {
		AND = { # slavery
			OR = { type = law_type:law_slavery_banned type = law_type:law_slave_trade type = law_type:law_legacy_slavery type = law_type:law_colonial_slavery type = law_type:law_debt_slavery }
			any_scope_amendment = {
				nr_amendment_is_ours = yes
				sponsor ?= { is_interest_group_type = ig_landowners }
			}
		}
		AND = { # policing
			OR = { type = law_type:law_no_police type = law_type:law_local_police type = law_type:law_dedicated_police type = law_type:law_militarized_police }
			any_scope_amendment = {
				nr_amendment_is_ours = yes
				sponsor ?= { is_interest_group_type = ig_landowners }
			}
		}
	}
}

# In event nr_amendment_room.1: this amendment's sponsor opposes the sponsor of the new one on this law. Scope: amendment.
nr_amendment_opposes_room = {
	nr_amendment_is_ours = yes
	OR = {
		AND = { # slavery
			scope:room_law = { OR = { type = law_type:law_slavery_banned type = law_type:law_slave_trade type = law_type:law_legacy_slavery type = law_type:law_colonial_slavery type = law_type:law_debt_slavery } }
			OR = {
				AND = {
					scope:room_ig = { is_interest_group_type = ig_devout }
					sponsor ?= { is_interest_group_type = ig_landowners }
				}
				AND = {
					scope:room_ig = { is_interest_group_type = ig_landowners }
					sponsor ?= { is_interest_group_type = ig_devout }
				}
			}
		}
		AND = { # policing
			scope:room_law = { OR = { type = law_type:law_no_police type = law_type:law_local_police type = law_type:law_dedicated_police type = law_type:law_militarized_police } }
			OR = {
				AND = {
					scope:room_ig = { is_interest_group_type = ig_devout }
					sponsor ?= { is_interest_group_type = ig_landowners }
				}
				AND = {
					scope:room_ig = { is_interest_group_type = ig_landowners }
					sponsor ?= { is_interest_group_type = ig_devout }
				}
			}
		}
	}
}
