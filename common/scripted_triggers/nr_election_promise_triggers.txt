# NR: promise to back the group's party at the next election (feature election_promise).
# Scope: country; in negotiation.1 scope:ig = the group, scope:law = the law being enacted.

# Option shown: elections that can be rigged, the ruler is not against the law.
nr_election_promise_available = {
	OR = { # elections: a voting law (vanilla country_has_voting_franchise also counts stray voting modifiers)
		has_law_or_variant = law_type:law_landed_voting
		has_law_or_variant = law_type:law_wealth_voting
		has_law_or_variant = law_type:law_census_voting
		has_law_or_variant = law_type:law_universal_suffrage
	}
	modifier:country_forbid_electoral_fraud_bool = no # no Tradition of Free Elections
	nr_ruler_not_against_law = yes
}

# ... but greyed out with the reason (show_as_unavailable).
nr_election_promise_blocked = {
	OR = {
		custom_tooltip = {
			text = nr_election_promise_already_tt
			scope:ig = { has_variable = nr_election_promise }
		}
		custom_tooltip = {
			text = nr_authority_cost_no_slot_tt
			nr_authority_cost_slot_free = no
		}
	}
}


# Rigging events (caciquismo.1 / .2), extra option for the devout: their promise waits for this campaign and their party
# (saved in nr_election_rigging_immediate) is neither listed by the event nor backed by an earlier extra option.
nr_election_promise_party_unlisted_devout = {
	exists = scope:nr_promised_party_devout
	NOR = {
		scope:party_1_scope ?= scope:nr_promised_party_devout
		scope:party_2_scope ?= scope:nr_promised_party_devout
		scope:party_3_scope ?= scope:nr_promised_party_devout
		scope:party_4_scope ?= scope:nr_promised_party_devout
		scope:party_5_scope ?= scope:nr_promised_party_devout
	}
}
