# NR: Devout church officials (negotiation option 2). Scope: country unless noted.

# AI (option negotiation.1.nr_o2_custom, via nr_neg_option_2_ai_*).
# Plenty of spare bureaucracy after the penalty.
nr_devout_officials_ai_good = {
	nr_campaign_spare_share >= 0.25
}

# Almost no spare bureaucracy left after the penalty.
nr_devout_officials_ai_bad = {
	nr_campaign_spare_share < 0.10
}

# ---------- Forms (event nr_devout_officials.1) ----------
# Every form has nr_devout_officials_<form>_available. A bureaucracy deficit is not checked: the player decides.

# The form is available, not offered yet in this event and not already running in another campaign
# (a second one would overwrite its modifier). $FORM$ = censorship / registers / ...
nr_devout_officials_can_offer = {
	nr_devout_officials_$FORM$_available = yes
	NOT = { has_variable = nr_devout_officials_offer_$FORM$ }
	NOT = { has_modifier = nr_devout_officials_$FORM$ }
}

nr_devout_officials_censorship_available = {
	has_law = law_type:law_right_of_assembly
	nr_ig_leader_not_against_law = { LAW = law_censorship }
}

nr_devout_officials_registers_available = {
	always = yes
}

nr_devout_officials_slaves_available = {
	NOT = { has_law = law_type:law_slavery_banned }
}

nr_devout_officials_church_tax_available = {
	OR = {
		has_law = law_type:law_state_religion
		has_law = law_type:law_millet_system
		has_law = law_type:law_people_of_the_book
		has_law = law_type:law_freedom_of_conscience
	}
}

nr_devout_officials_prisons_available = {
	NOT = { has_law = law_type:law_state_atheism }
}

nr_devout_officials_calendar_available = {
	NOT = { has_law = law_type:law_state_atheism }
}

nr_devout_officials_seminarists_available = {
	always = yes
}

nr_devout_officials_synod_available = {
	OR = {
		has_law = law_type:law_state_religion
		has_law = law_type:law_millet_system
		has_law = law_type:law_people_of_the_book
	}
	nr_ig_leader_not_against_law = { LAW = law_state_religion }
}
