# NR: general rule of the mod - an interest group never offers a form, petition or amendment that its leader is against.
# $IG$ = interest group type, $LAW$ = law key. True when the group or its leader does not exist. Scope: country.
nr_leader_not_against_law = {
	NOT = {
		ig:$IG$ ?= {
			leader ?= {
				law_stance = {
					law = law_type:$LAW$
					value < neutral
				}
			}
		}
	}
}
