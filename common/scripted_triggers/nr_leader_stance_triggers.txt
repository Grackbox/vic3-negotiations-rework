# NR: general rule of the mod - an interest group never offers a form, petition or amendment
# that its leader is against. Scope: country, scope:ig = the interest group.

# The leader's own stance on the law is not below neutral (true when there is no leader).
nr_ig_leader_not_against_law = {
	NOT = {
		scope:ig.leader ?= {
			law_stance = {
				law = law_type:$LAW$
				value < neutral
			}
		}
	}
}

# Same for the Devout leader, without scope:ig (usable in event triggers). Scope: country.
nr_devout_leader_not_against_law = {
	NOT = {
		ig:ig_devout ?= {
			leader ?= {
				law_stance = {

	}
}
