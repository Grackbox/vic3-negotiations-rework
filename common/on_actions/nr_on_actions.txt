# NR: additions to vanilla on_actions. Declared in our own file; the game merges on_action data across files
# (see vanilla common/on_actions/_on_actions.md). Vanilla on_actions only get our own on_actions appended via
# on_actions = { ... }; our effects live in our own nr_* on_actions, so nothing conflicts.

# Law enactment debate pool (same pool as vanilla education_laws.2 "Prussian education", weight 10)
on_law_checkpoint_debate = {
	random_events = {
		10 = nr_devout_chaplains.1
		10 = nr_devout_synod.1
		10 = nr_devout_church_tax.1
		10 = nr_devout_sunday.1
		10 = nr_devout_census.1
		10 = nr_devout_slavery.1
	}
}

# Root = law, owner = its country. Vanilla has an effect block here, so we only append our own on_action.
on_law_activated = {
	on_actions = { nr_on_law_activated }
}

# Root = country, enactment of any law begins (currently_enacting_law = the new law, the old law is still active).
on_law_enactment_started = {
	on_actions = { nr_on_law_enactment_started }
}

nr_on_law_enactment_started = {
	effect = {
		nr_petition_improve_stance = yes # groups with a petition for this law support its enactment
		nr_devout_petition_boost_all = yes # and their petition speeds it up at once
		nr_devout_church_tax_transfer = yes # church tax guarantee: amendment B goes into Freedom of Conscience / Total Separation
		nr_devout_census_attach_on_petition = yes # clergy petition for Appointed Bureaucrats: the Clerical Census comes with it
	}
}

nr_on_law_activated = {
	effect = {
		owner = {
			nr_devout_officials_check_censorship = yes # Censorship Committees end under Protected Speech
		}
		if = {
			limit = { has_amendment = amendment_type:amendment_nr_devout_fellow_believers }
			owner = { nr_devout_slavery_free_believers = yes } # a slavery law passed with the amendment
		}
	}
}

# Yearly pulse of every country: appended to the vanilla on_action.
on_yearly_pulse_country = {
	on_actions = { nr_on_yearly_pulse_country }
}

nr_on_yearly_pulse_country = {
	effect = {
		nr_devout_slavery_yearly = yes # "Do Not Enslave Fellow Believers": free the converted, calm the abolitionists
	}
}
