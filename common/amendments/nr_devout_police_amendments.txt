# NR: police amendments (feature devout_police), one per police law, different by design.
# Offered by nr_devout_police.2, 10 years after the Prison Ministry form of the church officials.
# Parent State Religion, sponsor Devout, 50 authority each: the state controls its own apparatus less.

amendment_nr_devout_parish_constables = {
	parent = law_state_religion

	allowed_laws = {
		law_local_police
	}

	modifier = {
		state_turmoil_effects_mult = -0.05 # order is kept by the parish
		interest_group_ig_devout_pol_str_mult = 0.05 # the parish, not the squire, chooses the constable
		interest_group_ig_landowners_pol_str_mult = -0.05 # takes back part of the landowners' bonus from local police
		country_authority_cost_add = 50 # the state controls its own apparatus less
	}

	possible = {
		always = yes
	}

	would_sponsor = {
		is_interest_group_type = ig_devout
		exists = scope:approval
		scope:approval > 0
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.25
}
