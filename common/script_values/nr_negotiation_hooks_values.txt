# NR: values used by the hooks in vanilla negotiations.

# Negotiation level (amenability_level 1 / 2 / 4) scaled down to 1 / 2 / 3. Used instead of the raw level for
# rewards to the group: the bribe (bribed_ig_benefits), soft bribe patronage, church officials. Requires scope:ig.
# Read explicitly from scope:ig, like vanilla: modifier multipliers are not always evaluated in the group's scope,
# and the vanilla amenability value errors out in country scope.
nr_neg_level_scale = {
	value = scope:ig.amenability_level
	max = 3
}

# Extra weight for option 1 (bribe, base 10) in set_neg_options: group attitude to bribes + leader traits.
# Like vanilla, option 1 is not guaranteed. Total weight never below 0.
nr_neg_option_1_modifier = {
	value = scope:ig.nr_bribe_ig_weight
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		add = scope:ig.nr_devout_softbribe_leader_weight
	}
	else = {
		add = scope:ig.nr_bribe_leader_weight
	}
	min = -10
}

# Extra weight for option 2 (bureaucracy, base 10) in set_neg_options.
nr_neg_option_2_modifier = {
	value = 0
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_devout } }
		add = scope:ig.nr_devout_officials_leader_weight
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		add = scope:ig.nr_landowners_offices_leader_weight
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		add = scope:ig.nr_industrialists_commissions_leader_weight
	}
}

# Extra weight for option 9 (SoL, vanilla base 5, Devout and Trade Unions +30). Landowners +15: paternalism is their ideology.
nr_neg_option_9_modifier = {
	value = 0
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_landowners } }
		add = 15
	}
	if = {
		limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
		add = 5 # model mill owners, but paternalism is not their creed
	}
}
