# NR: numbers for the Devout soft bribe - funds for the church's needs (negotiation option 1).

# Base weekly cost = vanilla bribe amount (neg_bribe_amount). Requires scope:ig.
nr_devout_softbribe_base_cost = {
	value = neg_bribe_amount
}

# Weight change of negotiation option 1 (base 10) from the Devout leader. Scope: interest group.
# Bold, prominent leaders raise money matters more often; modest, careful ones less often. Total never drops below 0.
nr_devout_softbribe_leader_weight = {
	value = 0
	if = {
		limit = { leader ?= { nr_bribe_leader_loves = yes } }
		add = 10 # Grifter / Expensive Tastes / Hedonist
	}
	if = {
		limit = { leader ?= { has_trait = ambitious } }
		add = 10
	}
	if = {
		limit = { leader ?= { has_trait = charismatic } }
		add = 10
	}
	if = {
		limit = { leader ?= { has_trait = arrogant } }
		add = 5
	}
	if = {
		limit = { leader ?= { prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD } }
		add = 10 # prominence 50+
	}
	if = {
		limit = { leader ?= { has_trait = reserved } }
		add = -5
	}
	if = {
		limit = { leader ?= { has_trait = cautious } }
		add = -5
	}
	min = -10 # base weight 10 + this >= 0
}

# Colonial missions: yearly share of each colonial subject state's population converted to the overlord's religion,
# and how many years it runs (same 10 years as the other forms).
nr_devout_softbribe_mission_share = 0.03
nr_devout_softbribe_mission_years = 10
