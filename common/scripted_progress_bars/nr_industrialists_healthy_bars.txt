# NR: the Industrialists' heavy industry promise: months in a row with enough paying levels (0-6). A month without
# them resets the bar. Evaluated in journal entry scope; the country (owner) holds copies of the type and the target.

nr_industrialists_healthy_bar = {
	name = "nr_industrialists_healthy_bar"
	desc = "nr_industrialists_healthy_bar_desc"

	default_green = yes

	monthly_progress = {
		if = {
			limit = {
				owner = {
					has_variable = nr_industrialists_healthy_target_c
					nr_industrialists_healthy_levels_c >= var:nr_industrialists_healthy_target_c
				}
			}
			add = {
				desc = "nr_industrialists_healthy_bar_ok"
				value = 1
			}
		}
		else = {
			add = {
				desc = "nr_industrialists_healthy_bar_reset"
				value = -6 # min_value 0 clamps it: the count starts again
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 6
}
