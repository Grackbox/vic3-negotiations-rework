nr_landowners_offices.11 = { # follow-up of patrimonial

	desc = nr_landowners_offices.11.desc

	duration = 3

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		exists = ig:ig_landowners
		has_variable = nr_landowners_offices_patrimonial_running
	}

	immediate = {
		ig:ig_landowners ?= { save_scope_as = ig }
	}

	option = { # amendment
		name = nr_landowners_offices.fu.amendment
		trigger = {
			nr_landowners_offices_patrimonial_justice_available = yes
		}
		custom_tooltip = nr_landowners_offices_lore_11_amendment
		nr_landowners_offices_add_patrimonial_justice = yes
	}

	option = { # decline (only while there was something to offer)
		name = nr_landowners_offices.fu.decline
		default_option = yes
		trigger = {
			nr_landowners_offices_patrimonial_justice_available = yes
		}
		scope:ig ?= {
			add_modifier = {
				name = nr_landowners_offices_declined
				days = normal_modifier_time
			}
		}
	}

	option = { # nothing to offer: neutral close
		name = nr_landowners_offices.fu.nothing
		default_option = yes
		trigger = {
			NOT = { nr_landowners_offices_patrimonial_justice_available = yes }
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { has_variable = nr_landowners_offices_patrimonial_running }
				remove_variable = nr_landowners_offices_patrimonial_running
			}
		}
	}
}
