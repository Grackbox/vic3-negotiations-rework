namespace = nr_devout_officials

# NR: where the church officials go (negotiation option 2 for the Devout, see nr_devout_officials_effects.txt).
# The base bureaucracy penalty (negotiation_bureaucracy x level) is already applied; forms rescale it
# (x0.75 / x1 / x1.25 / x1.5). All effects last 10 years and decay.
# Options shown: Clerical Advisers (always, default) + 2 random of the available forms:
# Censorship Committees (Right of Assembly), Parish Registers, The Church and the Slaves (slavery legal),
# Church Tax (state religion or freedom of conscience), Prisons, Calendar and Holidays (not state atheism),
# Seminarists in the Chancelleries, Synodal Administration (state religion).
# Forms that raise the penalty are only offered if they do not cause a bureaucracy deficit.

	placement = root
	title = nr_devout_officials.1.t
	desc = nr_devout_officials.1.desc
	flavor = nr_devout_officials.1.flavor

			leader ?= { save_scope_as = ig_leader }
		}
		# Pick 2 of the available forms at random
		random_list = {
			1 = {
				trigger = {
					nr_devout_officials_censorship_available = yes
				}
				set_variable = nr_devout_officials_offer_censorship
			}
			1 = {
				trigger = {
					always = yes
				}
				set_variable = nr_devout_officials_offer_registers

				trigger = {

				}
				set_variable = nr_devout_officials_offer_slaves

				trigger = {
					nr_devout_officials_church_tax_available = yes

				set_variable = nr_devout_officials_offer_church_tax

				trigger = {
					nr_devout_officials_not_state_atheism = yes

					nr_devout_officials_not_state_atheism = yes
				}

			1 = {
				trigger = {
					nr_devout_officials_censorship_available = yes

				set_variable = nr_devout_officials_offer_censorship
			}
			1 = {
				trigger = {
					always = yes
					NOT = { has_variable = nr_devout_officials_offer_registers }
				}
				set_variable = nr_devout_officials_offer_registers
			}

				trigger = {
					nr_devout_officials_slaves_available = yes
					NOT = { has_variable = nr_devout_officials_offer_slaves }
				}
				set_variable = nr_devout_officials_offer_slaves
			}
			1 = {
				trigger = {
					nr_devout_officials_church_tax_available = yes

				trigger = {
