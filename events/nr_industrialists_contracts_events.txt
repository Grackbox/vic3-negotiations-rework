	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		exists = ig:ig_industrialists
		has_variable = nr_industrialists_contracts_railway_running
	}

	immediate = {
		ig:ig_industrialists ?= { save_scope_as = ig }
	}

	option = { # amendment
		name = nr_industrialists_fu_amendment
		trigger = {
			nr_industrialists_guaranteed_return_amendment_available = yes
		}
		custom_tooltip = nr_industrialists_lore_amendment_guaranteed_return
		nr_industrialists_add_guaranteed_return = yes
	}

	option = { # decline
		name = nr_industrialists_fu_decline
		default_option = yes
		trigger = {
			nr_industrialists_guaranteed_return_amendment_available = yes
		}
		scope:ig ?= {
			add_modifier = {
				name = nr_industrialists_declined
				days = normal_modifier_time
			}
		}
	}

	option = { # nothing to offer: neutral close
		name = nr_industrialists_fu_nothing
		default_option = yes
		trigger = {
			NOT = { nr_industrialists_guaranteed_return_amendment_available = yes }
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { has_variable = nr_industrialists_contracts_railway_running }
				remove_variable = nr_industrialists_contracts_railway_running
			}
		}
	}
}
