namespace = nr_industrialists_enact

# NR: enactment events for the Industrialists' amendments (pattern of nr_landowners_enact).

nr_industrialists_enact.1 = { # amendment_nr_industrialists_guaranteed_return
	type = country_event
	placement = root
	title = nr_industrialists_enact.1.t
	desc = nr_industrialists_enact.1.desc
	flavor = nr_industrialists_enact.1.flavor

	category = enactment

	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		OR = {
			is_enacting_law = law_type:law_traditionalism
			is_enacting_law = law_type:law_interventionism
			is_enacting_law = law_type:law_agrarianism
			is_enacting_law = law_type:law_laissez_faire
		}
		exists = ig:ig_industrialists
		NOT = {
			nr_amendment_is_present = { AMENDMENT = amendment_nr_industrialists_guaranteed_return }
		}
		NOT = { has_modifier = nr_industrialists_contracts_railway } # the form is running: its follow-up offers the amendment
		nr_leader_not_against_law = { IG = ig_industrialists LAW = law_interventionism }
	}

	immediate = {
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
		ig:ig_industrialists ?= { save_scope_as = industrialists_ig }
		ig:ig_petty_bourgeoisie ?= { save_scope_as = other_ig }
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = { # refuse: the law passes faster
		name = nr_industrialists_enact.1.a
		default_option = yes
		custom_tooltip = nr_industrialists_enact_lore_a
		add_enactment_modifier = {
			name = nr_industrialists_enact_guaranteed_return_fast
		}
		scope:industrialists_ig ?= {
			add_modifier = {
				name = nr_industrialists_enact_guaranteed_return_rejected
				days = normal_modifier_time
			}
		}
	}

	option = { # attach the amendment
		name = nr_industrialists_enact.1.b
		custom_tooltip = nr_industrialists_enact_lore_b
		add_enactment_modifier = {
			name = nr_industrialists_enact_guaranteed_return_debate
		}
		nr_amendment_add_to_enacting_law = {
			AMENDMENT = amendment_nr_industrialists_guaranteed_return
			IG = ig_industrialists
		}
	}

	option = { # let another group decide
		name = nr_industrialists_enact.1.c
		custom_tooltip = nr_industrialists_enact_lore_c
		scope:other_ig ?= {
			add_modifier = {
				name = nr_industrialists_enact_guaranteed_return_heard
				days = normal_modifier_time
			}
		}
		scope:industrialists_ig ?= {
			add_modifier = {
				name = nr_industrialists_enact_guaranteed_return_rejected
				days = normal_modifier_time
			}
		}
	}
}
