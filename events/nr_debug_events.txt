# .4 - Devout regular bribe (negotiation option 1) without a negotiation: pay it as the option would, give the leader
#      a vice now (100%), or expose it now (generic_laws.2 needs a law being enacted).
nr_debug.4 = {
	type = country_event
	placement = root
	title = nr_debug.4.t
	desc = nr_debug.4.desc

	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		exists = ig:ig_devout
	}

	immediate = {
		ig:ig_devout = { save_scope_as = ig }
	}

	option = { # Pay the bribe as negotiation.1.o1 would
		name = nr_debug.4.a
		trigger = {
			nr_bribe_affordable = yes
			exists = currently_enacting_law # vanilla bribe values read the law being enacted
		}
		nr_bribe_pay = yes
		scope:ig = {
			add_modifier = {
				name = bribed_ig_benefits
				days = long_modifier_time
				multiplier = nr_neg_level_scale
				is_decaying = yes
			}
		}
		nr_bribe_exposure_roll = yes
	}

	option = { # The leader gets a vice now (100%)
		name = nr_debug.4.b
		scope:ig.leader ?= {
			nr_bribe_add_vice_trait = yes
		}
	}

	option = { # Expose the bribe now
		name = nr_debug.4.c
		trigger = {
			enacting_any_law = yes
		}
		trigger_event = { id = generic_laws.2 popup = yes }
	}

	option = {
		name = nr_debug.4.close
		default_option = yes
	}
}

		name = nr_debug.11.election
		trigger = {
			any_interest_group = {
				has_party = yes
				NOT = { has_variable = nr_election_promise }
			}
		}
		random_interest_group = {
			limit = {
				has_party = yes
				NOT = { has_variable = nr_election_promise }
			}
			save_scope_as = ig
			set_variable = { name = promise_quest_degree value = 2 }
			set_variable = nr_election_promise
			set_variable = { name = promise_quest_type value = 102 }
		}
		custom_tooltip = nr_election_promise_tt
		add_modifier = { name = active_ig_promise_country multiplier = 2 } # promise_quest without its law stance part
		scope:ig = {
			set_variable = ig_new_quest_starting
			add_modifier = { name = active_ig_promise }
		}
		if = {
			limit = { scope:ig = { is_interest_group_type = ig_armed_forces } }
			add_journal_entry = { type = je_nr_election_promise_armed_forces }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_devout } }
			add_journal_entry = { type = je_nr_election_promise_devout }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_industrialists } }
			add_journal_entry = { type = je_nr_election_promise_industrialists }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_intelligentsia } }
			add_journal_entry = { type = je_nr_election_promise_intelligentsia }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_landowners } }
			add_journal_entry = { type = je_nr_election_promise_landowners }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_petty_bourgeoisie } }
			add_journal_entry = { type = je_nr_election_promise_petty_bourgeoisie }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_rural_folk } }
			add_journal_entry = { type = je_nr_election_promise_rural_folk }
		}
		else_if = {
			limit = { scope:ig = { is_interest_group_type = ig_trade_unions } }
			add_journal_entry = { type = je_nr_election_promise_trade_unions }
		}
	}
