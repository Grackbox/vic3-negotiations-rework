namespace = nr_landowners_enact

# NR: enactment events for the Landowners' own amendments (pattern "IG amendment pair", like the Devout .1 events).
# In the vanilla on_law_checkpoint_debate pool. a (default): +15% enactment speed, Landowners -2 approval 5 years;
# b: attach the amendment, -15% speed; c: let another group decide - it gets +2, the Landowners -2.
# Not while the grant whose follow-up offers the same amendment is running.

nr_landowners_enact.1 = { # amendment_nr_landowners_noble_privilege
	type = country_event
	placement = root
	title = nr_landowners_enact.1.t
	desc = nr_landowners_enact.1.desc
	flavor = nr_landowners_enact.1.flavor

	category = enactment

	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		OR = {
			is_enacting_law = law_type:law_consumption_based_taxation
			is_enacting_law = law_type:law_land_based_taxation
			is_enacting_law = law_type:law_per_capita_based_taxation
			is_enacting_law = law_type:law_proportional_taxation
			is_enacting_law = law_type:law_graduated_taxation
		}
		exists = ig:ig_landowners
		NOT = {
			nr_amendment_is_present = { AMENDMENT = amendment_nr_landowners_noble_privilege }
		}
		NOT = { has_modifier = nr_landowners_grants_tax_exemptions } # the grant is running: its follow-up offers the amendment
		nr_landowners_leader_not_against_law = { LAW = law_land_based_taxation } # parent law of the amendment
	}

	immediate = {
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
		ig:ig_landowners ?= { save_scope_as = landowners_ig }
		ig:ig_petty_bourgeoisie ?= { save_scope_as = other_ig }
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = { # refuse: the law passes faster
		name = nr_landowners_enact.1.a
		default_option = yes
		custom_tooltip = nr_landowners_enact_lore_1a
		add_enactment_modifier = {
			name = nr_landowners_enact_privilege_fast
		}
		scope:landowners_ig ?= {
			add_modifier = {
				name = nr_landowners_enact_privilege_rejected
				days = normal_modifier_time
			}
		}
	}

	option = { # attach the amendment
		name = nr_landowners_enact.1.b
		custom_tooltip = nr_landowners_enact_lore_1b
		add_enactment_modifier = {
			name = nr_landowners_enact_privilege_debate
		}
		nr_amendment_add_to_enacting_law = {
			AMENDMENT = amendment_nr_landowners_noble_privilege
			IG = ig_landowners
		}
	}

	option = { # let another group decide
		name = nr_landowners_enact.1.c
		custom_tooltip = nr_landowners_enact_lore_1c
		scope:other_ig ?= {
			add_modifier = {
				name = nr_landowners_enact_privilege_heard
				days = normal_modifier_time
			}
		}
		scope:landowners_ig ?= {
			add_modifier = {
				name = nr_landowners_enact_privilege_rejected
				days = normal_modifier_time
			}
		}
	}
}
