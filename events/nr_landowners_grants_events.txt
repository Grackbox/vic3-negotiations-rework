namespace = nr_landowners_grants

# NR: landowners' grants (negotiation option 1, feature landowners_grants): the form of the grant.
# Base expenses (vanilla bribe amount, slot modifier) and the Landowners' favour are already applied.
# Options: Ranks and Orders (always, default, base effects only) + 2 random of the available forms. 10 years, decaying.
# Follow-ups .11 - .18 come 10 years later if the form was not interrupted by a law change.

nr_landowners_grants.1 = {
	type = country_event
	placement = root
	title = nr_landowners_grants.1.t
	desc = nr_landowners_grants.1.desc
	flavor = nr_landowners_grants.1.flavor

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	immediate = {
		ig:ig_landowners ?= {
			save_scope_as = ig
			leader ?= { save_scope_as = ig_leader }
		}
		# Pick 2 of the available forms at random
		nr_landowners_grants_offer_random_form = yes
		nr_landowners_grants_offer_random_form = yes
	}

	option = { # Ranks and Orders: base effects only, cost x1
		name = nr_landowners_grants.1.a
		default_option = yes
		custom_tooltip = nr_landowners_grants_lore_ranks
		custom_tooltip = nr_landowners_grants_ranks_tt
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = nr_landowners_grants.1.b
		trigger = {
			has_variable = nr_landowners_grants_offer_crown_lands
		}
		custom_tooltip = nr_landowners_grants_lore_crown_lands
		nr_landowners_grants_choose_form = { FORM = crown_lands FACTOR = 0.5 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.11
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = nr_landowners_grants.1.c
		trigger = {
			has_variable = nr_landowners_grants_offer_tax_exemptions
		}
		custom_tooltip = nr_landowners_grants_lore_tax_exemptions
		nr_landowners_grants_choose_form = { FORM = tax_exemptions FACTOR = 0.25 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.12
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { net_total_income < 0 }
				add = -10
			}
		}
	}

	option = {
		name = nr_landowners_grants.1.d
		trigger = {
			has_variable = nr_landowners_grants_offer_corn_laws
		}
		custom_tooltip = nr_landowners_grants_lore_corn_laws
		nr_landowners_grants_choose_form = { FORM = corn_laws FACTOR = 0.25 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.13
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { ig:ig_industrialists ?= { is_powerful = no } }
				add = 10
			}
		}
	}

	option = {
		name = nr_landowners_grants.1.e
		trigger = {
			has_variable = nr_landowners_grants_offer_land_bank
		}
		custom_tooltip = nr_landowners_grants_lore_land_bank
		nr_landowners_grants_choose_form = { FORM = land_bank FACTOR = 1.25 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.14
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { nr_landowners_grants_high_debt = yes }
				add = -10
			}
		}
	}

	option = {
		name = nr_landowners_grants.1.f
		trigger = {
			has_variable = nr_landowners_grants_offer_corvee
		}
		custom_tooltip = nr_landowners_grants_lore_corvee
		nr_landowners_grants_choose_form = { FORM = corvee FACTOR = 0.25 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.15
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = nr_landowners_grants.1.g
		trigger = {
			has_variable = nr_landowners_grants_offer_redemption
		}
		custom_tooltip = nr_landowners_grants_lore_redemption
		nr_landowners_grants_choose_form = { FORM = redemption FACTOR = 1.5 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.16
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = nr_landowners_grants.1.h
		trigger = {
			has_variable = nr_landowners_grants_offer_fugitives
		}
		custom_tooltip = nr_landowners_grants_lore_fugitives
		nr_landowners_grants_choose_form = { FORM = fugitives FACTOR = 0.75 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.17
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { any_political_movement = { is_political_movement_type = movement_slave_revolt } }
				add = 15
			}
		}
	}

	option = {
		name = nr_landowners_grants.1.i
		trigger = {
			has_variable = nr_landowners_grants_offer_slave_import
		}
		custom_tooltip = nr_landowners_grants_lore_slave_import
		nr_landowners_grants_choose_form = { FORM = slave_import FACTOR = 0.25 }
		hidden_effect = {
			trigger_event = {
				id = nr_landowners_grants.18
				days = 3650
				popup = yes
			}
		}
		ai_chance = {
			base = 10
		}
	}

	after = {
		hidden_effect = {
			nr_landowners_grants_cleanup = yes
		}
	}
}

nr_landowners_grants.11 = {
	type = country_event
	placement = root
	title = nr_landowners_grants.11.t
	desc = nr_landowners_grants.11.desc
	flavor = nr_landowners_grants.11.flavor

	duration = 3

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		exists = ig:ig_landowners
		has_variable = nr_landowners_grants_crown_lands_running
	}

	immediate = {
		ig:ig_landowners ?= { save_scope_as = ig }
	}

	option = { # petition
		name = nr_landowners_grants.11.a
		trigger = {
			nr_landowners_tenant_petition_available = yes
		}
		custom_tooltip = nr_landowners_grants_lore_11a
		custom_tooltip = nr_landowners_tenant_petition_tt
		add_journal_entry = { type = je_nr_landowners_tenant_petition }
	}

	option = { # Already in place: thanks
		name = nr_landowners_grants.11.d
		default_option = yes
		trigger = {
			nr_landowners_tenant_content = yes
		}
		custom_tooltip = nr_landowners_grants_lore_11d
		scope:ig ?= {
			add_modifier = {
				name = nr_landowners_grants_content
				days = normal_modifier_time
			}
		}
	}

	option = { # Decline (only while there was something to offer)
		name = nr_landowners_grants.11.c
		default_option = yes
		trigger = {
			nr_landowners_tenant_content = no
			nr_landowners_tenant_petition_available = yes
		}
		custom_tooltip = nr_landowners_grants_lore_11c
		scope:ig ?= {
			add_modifier = {
				name = nr_landowners_grants_declined
				days = normal_modifier_time
			}
		}
	}

	option = { # Nothing to offer: neutral close
		name = nr_landowners_grants.11.e
		default_option = yes
		trigger = {
			nr_landowners_tenant_content = no
			nr_landowners_tenant_petition_available = no
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { has_variable = nr_landowners_grants_crown_lands_running }
				remove_variable = nr_landowners_grants_crown_lands_running
			}
		}
	}
}
