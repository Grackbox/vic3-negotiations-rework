namespace = nr_devout_synod

# NR: follow-up of the Synodal Administration form (feature devout_synod).
# .1 - enactment of State Religion or its variants: may attach the amendment to the law being enacted.
# .2 - 10 years after the form was chosen: if a state church law is still active and the amendment is absent,
#      offer to write the synod into the law; otherwise no event.

nr_devout_synod.1 = {
	type = country_event
	placement = root
	title = nr_devout_synod.1.t
	desc = nr_devout_synod.1.desc
	flavor = nr_devout_synod.1.flavor

	category = enactment

	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		OR = {
			is_enacting_law = law_type:law_state_religion
			is_enacting_law = law_type:law_millet_system
			is_enacting_law = law_type:law_people_of_the_book
		}
		exists = ig:ig_devout
		NOT = {
			nr_amendment_is_present = { AMENDMENT = amendment_nr_devout_synod }
		}
		NOT = { has_modifier = nr_devout_officials_synod } # the synod campaign is running: its follow-up offers the amendment
		nr_leader_not_against_law = { IG = ig_devout LAW = law_state_religion } # parent law of the amendment
	}

	immediate = {
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
		ig:ig_devout ?= { save_scope_as = devout_ig }
		ig:ig_intelligentsia ?= { save_scope_as = intelligentsia_ig }
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = { # The church governs itself
		name = nr_devout_synod.1.a
		default_option = yes
		custom_tooltip = nr_devout_synod_lore_1a
		add_enactment_modifier = {
			name = nr_devout_synod_self_rule
		}
		scope:devout_ig ?= {
			add_modifier = {
				name = nr_devout_synod_rejected
				days = normal_modifier_time
			}
		}
	}

	option = { # Write the synod into the law being enacted
		name = nr_devout_synod.1.b
		custom_tooltip = nr_devout_synod_lore_1b
		add_enactment_modifier = {
			name = nr_devout_synod_debate
		}
		nr_amendment_add_to_enacting_law = {
			AMENDMENT = amendment_nr_devout_synod
			IG = ig_devout
		}
	}

	option = { # Let the intelligentsia decide
		name = nr_devout_synod.1.c
		custom_tooltip = nr_devout_synod_lore_1c
		scope:intelligentsia_ig ?= {
			add_modifier = {
				name = nr_devout_synod_intelligentsia_heard
				days = normal_modifier_time
			}
		}
		scope:devout_ig ?= {
			add_modifier = {
				name = nr_devout_synod_rejected
				days = normal_modifier_time
			}
		}
	}
}

nr_devout_synod.2 = {
	type = country_event
	placement = root
	title = nr_devout_synod.2.t
	desc = nr_devout_synod.2.desc
	flavor = nr_devout_synod.2.flavor

	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		exists = ig:ig_devout
		nr_devout_synod_amendment_available = yes
	}

	immediate = {
		ig:ig_devout ?= {
			save_scope_as = ig
			save_scope_as = devout_ig
		}
	}

	option = { # Write the synod into the law
		name = nr_devout_synod.2.a
		custom_tooltip = nr_devout_synod_lore_2a
		nr_devout_synod_add_amendment = yes
	}

	option = { # Decline
		name = nr_devout_synod.2.b
		default_option = yes
		custom_tooltip = nr_devout_synod_lore_2b
		scope:devout_ig ?= {
			add_modifier = {
				name = nr_devout_synod_declined
				days = normal_modifier_time
			}
		}
	}
}
